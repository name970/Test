//! Shared Plauthium primitives.
//!
//! Every consensus quantity is an integer (spec §1.4): amounts are [`Atoms`],
//! field arithmetic is over [`Mersenne61`], and every committed object is a
//! 512-bit [`DomainSeparatedHash`] of a typed, length-prefixed encoding.
//!
//! **PoC status:** [`DomainSeparatedHash`] and [`MockXof`] are deterministic,
//! *non-cryptographic* placeholders for SHA3-512 and SHAKE256. They keep the
//! normative interface (label, injective encoding, 64-byte digests) so that the
//! real primitives can be swapped in without touching callers.

use core::fmt;
use core::ops::{Add, AddAssign, Mul, Neg, Sub};

// ---------------------------------------------------------------------------
// Amounts and supply (spec §7.6, eq. 7.18)
// ---------------------------------------------------------------------------

/// Number of atoms in one PLAU (`1 TKN = 10^8 atoms`).
pub const ATOMS_PER_PLAU: u64 = 100_000_000;

/// Fixed supply in whole PLAU.
pub const TOTAL_SUPPLY_PLAU: u64 = 15_000_000;

/// Fixed supply `M = 1.5 * 10^15` atoms. It never changes in any reachable state.
pub const TOTAL_SUPPLY: Atoms = Atoms(1_500_000_000_000_000);

// `M = 15,000,000 PLAU` and `M < 2^51 < p`; evaluated at compile time, so an
// overflow or a violated bound fails the build.
const _: () = assert!(TOTAL_SUPPLY_PLAU * ATOMS_PER_PLAU == TOTAL_SUPPLY.0);
const _: () = assert!(TOTAL_SUPPLY.0 < (1u64 << 51));
const _: () = assert!((1u64 << 51) < Mersenne61::P);

/// An integer amount of the single native asset, in atoms.
#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct Atoms(pub u64);

impl Atoms {
    /// Zero atoms.
    pub const ZERO: Atoms = Atoms(0);

    /// Converts whole PLAU to atoms, or `None` on overflow.
    pub const fn from_plau(plau: u64) -> Option<Atoms> {
        match plau.checked_mul(ATOMS_PER_PLAU) {
            Some(atoms) => Some(Atoms(atoms)),
            None => None,
        }
    }

    /// The raw atom count.
    pub const fn get(self) -> u64 {
        self.0
    }

    /// Checked addition.
    pub const fn checked_add(self, rhs: Atoms) -> Option<Atoms> {
        match self.0.checked_add(rhs.0) {
            Some(v) => Some(Atoms(v)),
            None => None,
        }
    }

    /// Checked subtraction.
    pub const fn checked_sub(self, rhs: Atoms) -> Option<Atoms> {
        match self.0.checked_sub(rhs.0) {
            Some(v) => Some(Atoms(v)),
            None => None,
        }
    }
}

impl fmt::Display for Atoms {
    /// Formats as PLAU with all eight decimal places, e.g. `10.00000000 PLAU`.
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(
            f,
            "{}.{:08} PLAU",
            self.0 / ATOMS_PER_PLAU,
            self.0 % ATOMS_PER_PLAU
        )
    }
}

/// Integer Fresh Work share weight `w_s` (spec §6.1).
///
/// **PoC width:** normatively `w_s = floor(2^640 / Θ_S(s))`, an integer of up
/// to 640 bits. The PoC narrows it to `u64` so that every product with a slot
/// counter (D'Hondt) or an atom budget (fee-lot payouts) is exact in `u128`.
pub type ShareWeight = u64;

// ---------------------------------------------------------------------------
// Base field F_p, p = 2^61 - 1 (spec §3.1, Table 1)
// ---------------------------------------------------------------------------

/// An element of the base field `F_p`, `p = 2^61 - 1`, kept canonical in `[0, p)`.
#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct Mersenne61(u64);

impl Mersenne61 {
    /// The Mersenne prime `p = 2^61 - 1 = 2,305,843,009,213,693,951`.
    pub const P: u64 = (1u64 << 61) - 1;
    /// Additive identity.
    pub const ZERO: Mersenne61 = Mersenne61(0);
    /// Multiplicative identity.
    pub const ONE: Mersenne61 = Mersenne61(1);

    /// Reduces any `u64` into the field.
    pub const fn new(value: u64) -> Self {
        Mersenne61(Self::reduce_u128(value as u128))
    }

    /// Centered lift of a signed integer: negative values map to `p - |v| mod p`.
    pub const fn from_i64(value: i64) -> Self {
        let magnitude = Self::new(value.unsigned_abs());
        if value < 0 {
            magnitude.neg_const()
        } else {
            magnitude
        }
    }

    /// The canonical representative in `[0, p)`.
    pub const fn value(self) -> u64 {
        self.0
    }

    /// Reduces any `u128` modulo `p` using `2^61 = 1 (mod p)`.
    pub const fn reduce_u128(x: u128) -> u64 {
        let p = Self::P as u128;
        // x < 2^128  =>  folded < 2^61 + 2^67  =>  refolded < 2^61 + 2^7 < 2p.
        let folded = (x & p) + (x >> 61);
        let refolded = ((folded & p) + (folded >> 61)) as u64;
        if refolded >= Self::P {
            refolded - Self::P
        } else {
            refolded
        }
    }

    const fn neg_const(self) -> Self {
        if self.0 == 0 {
            self
        } else {
            Mersenne61(Self::P - self.0)
        }
    }
}

impl Add for Mersenne61 {
    type Output = Self;
    fn add(self, rhs: Self) -> Self {
        // Both operands < 2^61, so the sum < 2^62 cannot overflow.
        Mersenne61::new(self.0 + rhs.0)
    }
}

impl AddAssign for Mersenne61 {
    fn add_assign(&mut self, rhs: Self) {
        *self = *self + rhs;
    }
}

impl Sub for Mersenne61 {
    type Output = Self;
    fn sub(self, rhs: Self) -> Self {
        self + (-rhs)
    }
}

impl Neg for Mersenne61 {
    type Output = Self;
    fn neg(self) -> Self {
        self.neg_const()
    }
}

impl Mul for Mersenne61 {
    type Output = Self;
    fn mul(self, rhs: Self) -> Self {
        Mersenne61(Self::reduce_u128(self.0 as u128 * rhs.0 as u128))
    }
}

// Domain-separated hashing (spec §1.4, eq. 1.3) — PLACEHOLDER for SHA3-512

/// Digest length of SHA3-512 in bytes.
pub const DIGEST_LEN: usize = 64;

/// Versioned domain labels fixed by the specification.
pub mod labels {
    /// Fresh Work seed, eq. (5.2).
    pub const FW_INPUT: &str = "FW/INPUT/v2";
    /// Expansion domain of the left operand `X`, eq. (5.3).
    pub const FW_A: &str = "FW/A";
    /// Expansion domain of the right operand `Y`, eq. (5.3).
    pub const FW_B: &str = "FW/B";
    /// Commitment to the canonical product `XY`, eq. (5.4).
    pub const FW_OUTPUT: &str = "FW/OUTPUT/v2";
    /// Work identifier, which is also the ShareID, eq. (5.4).
    pub const FW_ID: &str = "FW/ID/v2";
    /// Identifier of the block an attempt would seal, eq. (5.4).
    pub const BLOCK: &str = "BLOCK/v2";
    /// Share lottery value, eq. (5.5).
    pub const FW_SHARE: &str = "FW/SHARE/v2";
    /// Seal lottery value, eq. (5.5).
    pub const FW_SEAL: &str = "FW/SEAL/v2";
    /// Transaction effect identifier, eq. (7.5).
    pub const TX_EFFECTS: &str = "TX/EFFECTS/v2";
    /// Signature context string, eqs. (2.3)–(2.5).
    pub const PLAU_TX: &str = "PLAU/TX/v2";
    /// Freivalds challenge domain. The spec requires one domain per round
    /// (§3.2) but does not name it; this label is assigned by the PoC.
    pub const FW_FREIVALDS: &str = "FW/FREIVALDS/v2";
}

/// A 512-bit digest `H_512(Enc(LABEL/v2, f_1, ..., f_k))`.
///
/// Ordering is lexicographic over the 64 bytes, which equals the numeric order
/// of the digest read as a big-endian integer (spec §1.4) and the canonical
/// 64-byte-string order used for ShareID tie-breaks (spec §6.3).
///
/// **PLACEHOLDER:** the compression function is a non-cryptographic mixer.
/// Replace with SHA3-512 (FIPS 202) before any security claim is made.
#[derive(Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct DomainSeparatedHash([u8; DIGEST_LEN]);

impl DomainSeparatedHash {
    /// Hashes `fields` under `label` with the injective encoding of [`DomainHasher`].
    pub fn new(label: &str, fields: &[&[u8]]) -> Self {
        let mut hasher = DomainHasher::new(label);
        for field in fields {
            hasher.field(field);
        }
        hasher.finalize()
    }

    /// Wraps raw digest bytes.
    pub const fn from_bytes(bytes: [u8; DIGEST_LEN]) -> Self {
        DomainSeparatedHash(bytes)
    }

    /// The digest bytes, big-endian.
    pub const fn as_bytes(&self) -> &[u8; DIGEST_LEN] {
        &self.0
    }

    /// The digest bytes by value; inverse of [`DomainSeparatedHash::from_bytes`].
    pub const fn to_bytes(&self) -> [u8; DIGEST_LEN] {
        self.0
    }

    /// Full lowercase hex encoding (128 characters).
    pub fn to_hex(&self) -> String {
        self.0.iter().map(|b| format!("{b:02x}")).collect()
    }

    /// First `bytes` bytes as hex, for logs and summaries.
    pub fn short_hex(&self, bytes: usize) -> String {
        self.0[..bytes.min(DIGEST_LEN)]
            .iter()
            .map(|b| format!("{b:02x}"))
            .collect()
    }
}

impl fmt::Debug for DomainSeparatedHash {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "H512({}..)", self.short_hex(8))
    }
}

/// Incremental builder for `H_512(Enc(label, f_1, ..., f_k))`.
///
/// `Enc` is injective: the label and every field are prefixed by their byte
/// length as a big-endian `u64`, and the field count is bound at finalization.
/// Field order is fixed by each call site, as the spec requires.
#[derive(Clone)]
pub struct DomainHasher {
    sponge: MockSponge,
    fields: u64,
}

impl DomainHasher {
    /// Starts a hash under a versioned domain label.
    pub fn new(label: &str) -> Self {
        let mut sponge = MockSponge::new();
        sponge.absorb(&(label.len() as u64).to_be_bytes());
        sponge.absorb(label.as_bytes());
        DomainHasher { sponge, fields: 0 }
    }

    /// Appends one length-prefixed byte field.
    pub fn field(&mut self, bytes: &[u8]) -> &mut Self {
        self.begin_field(bytes.len());
        self.sponge.absorb(bytes);
        self
    }

    /// Appends a `u64` field (8 bytes, big-endian).
    pub fn field_u64(&mut self, value: u64) -> &mut Self {
        self.field(&value.to_be_bytes())
    }

    /// Appends a digest field.
    pub fn field_digest(&mut self, digest: &DomainSeparatedHash) -> &mut Self {
        self.field(digest.as_bytes())
    }

    /// Appends a field of `i8` values, one byte each (two's complement).
    pub fn field_i8s(&mut self, values: &[i8]) -> &mut Self {
        self.begin_field(values.len());
        for &v in values {
            self.sponge.absorb_byte(v as u8);
        }
        self
    }

    /// Appends a field of `i32` values, four big-endian bytes each.
    pub fn field_i32s(&mut self, values: &[i32]) -> &mut Self {
        self.begin_field(values.len() * 4);
        for &v in values {
            self.sponge.absorb(&v.to_be_bytes());
        }
        self
    }

    /// Completes the encoding and returns the digest.
    pub fn finalize(&self) -> DomainSeparatedHash {
        let mut sponge = self.sponge.clone();
        sponge.absorb(&self.fields.to_be_bytes());
        DomainSeparatedHash(sponge.squeeze())
    }

    fn begin_field(&mut self, byte_len: usize) {
        self.fields += 1;
        self.sponge.absorb(&(byte_len as u64).to_be_bytes());
    }
}

/// Extendable-output stream bound to a label and a seed.
///
/// **PLACEHOLDER** for `SHAKE256(Enc(label, seed))`; output block `i` is
/// `H_512(Enc(label, seed, i))` of the placeholder hash.
#[derive(Clone)]
pub struct MockXof {
    prefix: DomainHasher,
    counter: u64,
    block: [u8; DIGEST_LEN],
    offset: usize,
}

impl MockXof {
    /// Starts a stream for `label` and `seed`.
    pub fn new(label: &str, seed: &DomainSeparatedHash) -> Self {
        let mut prefix = DomainHasher::new(label);
        prefix.field_digest(seed);
        MockXof {
            prefix,
            counter: 0,
            block: [0; DIGEST_LEN],
            offset: DIGEST_LEN,
        }
    }

    /// Next output byte.
    pub fn next_byte(&mut self) -> u8 {
        if self.offset == DIGEST_LEN {
            let mut block = self.prefix.clone();
            block.field_u64(self.counter);
            self.block = *block.finalize().as_bytes();
            self.counter += 1;
            self.offset = 0;
        }
        let byte = self.block[self.offset];
        self.offset += 1;
        byte
    }

    /// Next eight output bytes as a big-endian `u64`.
    pub fn next_u64(&mut self) -> u64 {
        let mut bytes = [0u8; 8];
        for b in &mut bytes {
            *b = self.next_byte();
        }
        u64::from_be_bytes(bytes)
    }
}

const SPONGE_LANES: usize = 8;
const GOLDEN_GAMMA: u64 = 0x9e37_79b9_7f4a_7c15;

/// Non-cryptographic 512-bit state used by the hashing placeholders.
#[derive(Clone)]
struct MockSponge {
    lanes: [u64; SPONGE_LANES],
    buffer: [u8; 8],
    buffered: usize,
    absorbed: u64,
}

impl MockSponge {
    fn new() -> Self {
        let mut lanes = [0u64; SPONGE_LANES];
        for (i, lane) in lanes.iter_mut().enumerate() {
            *lane = mix64(GOLDEN_GAMMA.wrapping_mul(i as u64 + 1));
        }
        MockSponge {
            lanes,
            buffer: [0; 8],
            buffered: 0,
            absorbed: 0,
        }
    }

    fn absorb(&mut self, bytes: &[u8]) {
        for &b in bytes {
            self.absorb_byte(b);
        }
    }

    fn absorb_byte(&mut self, byte: u8) {
        self.buffer[self.buffered] = byte;
        self.buffered += 1;
        self.absorbed = self.absorbed.wrapping_add(1);
        if self.buffered == 8 {
            self.absorb_word(u64::from_le_bytes(self.buffer));
            self.buffered = 0;
        }
    }

    fn absorb_word(&mut self, word: u64) {
        for i in 0..SPONGE_LANES {
            let neighbour = self.lanes[(i + SPONGE_LANES - 1) % SPONGE_LANES];
            self.lanes[i] = mix64(self.lanes[i] ^ word ^ neighbour.rotate_left(17));
        }
    }

    fn squeeze(mut self) -> [u8; DIGEST_LEN] {
        let absorbed = self.absorbed;
        self.absorb_byte(0x80);
        while self.buffered != 0 {
            self.absorb_byte(0);
        }
        self.absorb_word(absorbed);
        for _ in 0..SPONGE_LANES {
            self.absorb_word(GOLDEN_GAMMA);
        }
        let mut out = [0u8; DIGEST_LEN];
        for (chunk, lane) in out.chunks_exact_mut(8).zip(self.lanes) {
            chunk.copy_from_slice(&lane.to_be_bytes());
        }
        out
    }
}

/// SplitMix64 finalizer.
const fn mix64(mut z: u64) -> u64 {
    z = (z ^ (z >> 30)).wrapping_mul(0xbf58_476d_1ce4_e5b9);
    z = (z ^ (z >> 27)).wrapping_mul(0x94d0_49bb_1331_11eb);
    z ^ (z >> 31)
}

/// The five-component block resource vector `(B_bytes, B_sig, B_ZK, B_state, B_DA)`
/// of eq. (7.31). A block is valid only if the sum of its transactions' vectors
/// is componentwise at most the block limit.
#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
pub struct BlockResourceVector {
    /// Serialized bytes, including post-quantum signatures and keys.
    pub bytes: u64,
    /// Signature verifications.
    pub signature_checks: u64,
    /// Zero-knowledge proof verifications.
    pub zk_checks: u64,
    /// State writes.
    pub state_writes: u64,
    /// Data-availability payload bytes.
    pub da_bytes: u64,
}

impl BlockResourceVector {
    /// The zero vector.
    pub const ZERO: BlockResourceVector = BlockResourceVector {
        bytes: 0,
        signature_checks: 0,
        zk_checks: 0,
        state_writes: 0,
        da_bytes: 0,
    };

    /// Componentwise checked sum.
    pub fn checked_add(&self, other: &Self) -> Option<Self> {
        Some(BlockResourceVector {
            bytes: self.bytes.checked_add(other.bytes)?,
            signature_checks: self.signature_checks.checked_add(other.signature_checks)?,
            zk_checks: self.zk_checks.checked_add(other.zk_checks)?,
            state_writes: self.state_writes.checked_add(other.state_writes)?,
            da_bytes: self.da_bytes.checked_add(other.da_bytes)?,
        })
    }

    /// The componentwise order `self ≼ limit`.
    pub fn fits_within(&self, limit: &Self) -> bool {
        self.bytes <= limit.bytes
            && self.signature_checks <= limit.signature_checks
            && self.zk_checks <= limit.zk_checks
            && self.state_writes <= limit.state_writes
            && self.da_bytes <= limit.da_bytes
    }

    /// Sums transaction vectors and checks the block limit; `None` if any
    /// component overflows or exceeds its limit.
    pub fn accumulate<'a>(
        txs: impl IntoIterator<Item = &'a BlockResourceVector>,
        limit: &Self,
    ) -> Option<Self> {
        let total = txs
            .into_iter()
            .try_fold(Self::ZERO, |acc, tx| acc.checked_add(tx))?;
        total.fits_within(limit).then_some(total)
    }
}

//! Module 1 — the canonical Fresh Work tile and its Freivalds verification.
//!
//! The native work unit is the exact integer product `Z = XY` of two
//! `512 x 512` INT8 matrices with entries in `{-127, ..., 127}` (spec §3.1).
//! Every entry of `Z` satisfies `|Z_ij| <= 512 * 127^2 = 8,258,048 < 2^23`
//! (eq. 3.2), so two admissible results that agree modulo `p = 2^61 - 1` are
//! equal over the integers (eq. 3.3). Correctness is checked by `r_F = 6`
//! Freivalds rounds over `F_p` (§3.2), whose static error is at most `p^-6`.

use core::fmt;

use primitives::{DomainHasher, DomainSeparatedHash, Mersenne61, MockXof, labels};

/// Tile dimension `n = 512`.
pub const TILE_DIM: usize = 512;
/// Entries per tile, `n^2`.
pub const TILE_ELEMS: usize = TILE_DIM * TILE_DIM;
/// Smallest admissible INT8 input. `-128` is excluded: `512 * 128^2 = 2^23`
/// would break the strict bound of eq. (3.2).
pub const INT8_MIN: i8 = -127;
/// Largest admissible INT8 input.
pub const INT8_MAX: i8 = 127;
/// Maximum absolute value of an accumulated INT32 entry, `512 * 127^2`.
pub const ACCUMULATION_BOUND: i32 = 8_258_048;

/// Normative number of Freivalds rounds, `r_F = 6`.
pub const DEFAULT_FREIVALDS_ROUNDS: usize = 6;
/// Classical multiply-accumulate count of one tile, `n^3 = 2^27` (eq. 3.4).
pub const CLASSICAL_MACS: u64 = (TILE_DIM as u64) * (TILE_DIM as u64) * (TILE_DIM as u64);
/// Field multiplications of six Freivalds rounds, `3 r_F n^2 = 4,718,592` (eq. 3.8).
pub const FREIVALDS_FIELD_MULS: u64 = 3 * DEFAULT_FREIVALDS_ROUNDS as u64 * TILE_ELEMS as u64;

const _: () = assert!(TILE_DIM as i64 * 127 * 127 == ACCUMULATION_BOUND as i64);
const _: () = assert!((ACCUMULATION_BOUND as i64) < (1 << 23));
const _: () = assert!(TILE_DIM as i64 * 128 * 128 == 1 << 23);
const _: () = assert!(CLASSICAL_MACS == 1 << 27);
const _: () = assert!(FREIVALDS_FIELD_MULS * 256 == 9 * CLASSICAL_MACS);

/// Matrix entry types admitted by the tile domain.
pub trait TileElement: Copy + Default + Into<i64> {
    /// Largest admissible absolute value.
    const MAX_ABS: i64;
}

impl TileElement for i8 {
    const MAX_ABS: i64 = INT8_MAX as i64;
}

impl TileElement for i32 {
    const MAX_ABS: i64 = ACCUMULATION_BOUND as i64;
}

/// A `512 x 512` row-major matrix whose entries are always inside the tile domain.
#[derive(Clone, PartialEq, Eq)]
pub struct Matrix<T> {
    data: Box<[T]>,
}

/// The canonical Work Unit: a `512 x 512` matrix over `{-127, ..., 127}`.
pub type WorkUnit = Matrix<i8>;
/// The accumulation target: a `512 x 512` INT32 matrix with `|z| <= 8,258,048`.
pub type AccumulationTarget = Matrix<i32>;

/// Rejection reasons for matrices outside the tile domain.
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum WorkError {
    /// The entry count is not `512 * 512`.
    Dimension {
        /// Required entry count.
        expected: usize,
        /// Supplied entry count.
        found: usize,
    },
    /// An entry lies outside the admissible range.
    OutOfDomain {
        /// Row of the offending entry.
        row: usize,
        /// Column of the offending entry.
        col: usize,
        /// The offending value.
        value: i64,
        /// The admissible absolute bound.
        max_abs: i64,
    },
}

impl fmt::Display for WorkError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            WorkError::Dimension { expected, found } => {
                write!(f, "tile needs {expected} entries, got {found}")
            }
            WorkError::OutOfDomain {
                row,
                col,
                value,
                max_abs,
            } => write!(
                f,
                "entry ({row}, {col}) = {value} is outside [-{max_abs}, {max_abs}]"
            ),
        }
    }
}

impl std::error::Error for WorkError {}

impl<T: TileElement> Matrix<T> {
    /// Builds a tile from row-major entries, rejecting wrong sizes and out-of-domain values.
    pub fn from_vec(data: Vec<T>) -> Result<Self, WorkError> {
        if data.len() != TILE_ELEMS {
            return Err(WorkError::Dimension {
                expected: TILE_ELEMS,
                found: data.len(),
            });
        }
        if let Some(index) = data.iter().position(|&v| !in_domain(v)) {
            return Err(out_of_domain(index, data[index]));
        }
        Ok(Matrix {
            data: data.into_boxed_slice(),
        })
    }

    /// The all-zero tile.
    pub fn zeros() -> Self {
        Matrix {
            data: vec![T::default(); TILE_ELEMS].into_boxed_slice(),
        }
    }

    /// Entry at `(row, col)`. Panics if either index is `>= 512`.
    pub fn get(&self, row: usize, col: usize) -> T {
        self.data[index(row, col)]
    }

    /// Replaces the entry at `(row, col)` if the new value is in the domain.
    pub fn set(&mut self, row: usize, col: usize, value: T) -> Result<(), WorkError> {
        let i = index(row, col);
        if !in_domain(value) {
            return Err(out_of_domain(i, value));
        }
        self.data[i] = value;
        Ok(())
    }

    /// Row `row` as a slice.
    pub fn row(&self, row: usize) -> &[T] {
        &self.data[row * TILE_DIM..(row + 1) * TILE_DIM]
    }

    /// All entries, row-major.
    pub fn as_slice(&self) -> &[T] {
        &self.data
    }

    /// Largest absolute entry.
    pub fn max_abs(&self) -> i64 {
        self.data.iter().map(|&v| v.into().abs()).max().unwrap_or(0)
    }
}

impl Matrix<i8> {
    /// The identity tile.
    pub fn identity() -> Self {
        let mut m = Self::zeros();
        for i in 0..TILE_DIM {
            m.data[index(i, i)] = 1;
        }
        m
    }

    /// The exact INT32 product `self * rhs` (`n^3` multiply-accumulates).
    ///
    /// No partial sum can leave `[-8,258,048, 8,258,048]`, so the result is in
    /// the accumulation domain by construction.
    pub fn product(&self, rhs: &Matrix<i8>) -> Matrix<i32> {
        let mut out = vec![0i32; TILE_ELEMS];
        for (out_row, lhs_row) in out
            .chunks_exact_mut(TILE_DIM)
            .zip(self.data.chunks_exact(TILE_DIM))
        {
            for (&a, rhs_row) in lhs_row.iter().zip(rhs.data.chunks_exact(TILE_DIM)) {
                if a == 0 {
                    continue;
                }
                let a = i32::from(a);
                for (o, &b) in out_row.iter_mut().zip(rhs_row) {
                    *o += a * i32::from(b);
                }
            }
        }
        Matrix {
            data: out.into_boxed_slice(),
        }
    }
}

impl<T: TileElement + fmt::Debug> fmt::Debug for Matrix<T> {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(
            f,
            "Matrix<{}>[{TILE_DIM}x{TILE_DIM}; row0[..4]={:?}]",
            core::any::type_name::<T>(),
            &self.data[..4]
        )
    }
}

fn in_domain<T: TileElement>(value: T) -> bool {
    value.into().abs() <= T::MAX_ABS
}

fn out_of_domain<T: TileElement>(index: usize, value: T) -> WorkError {
    WorkError::OutOfDomain {
        row: index / TILE_DIM,
        col: index % TILE_DIM,
        value: value.into(),
        max_abs: T::MAX_ABS,
    }
}

fn index(row: usize, col: usize) -> usize {
    assert!(row < TILE_DIM && col < TILE_DIM, "tile index out of range");
    row * TILE_DIM + col
}

impl FreshWorkContext {
    /// The injective encoding `Enc(x)` of all sixteen fields in normative order.
    pub fn encode(&self) -> Vec<u8> {
        let fields: [&[u8]; FRESH_WORK_CONTEXT_FIELDS] = [
            &self.chain_id.to_be_bytes(),
            self.genesis_id.as_bytes(),
            &self.protocol_version.to_be_bytes(),
            &self.crypto_suite_id.to_be_bytes(),
            self.program_id.as_bytes(),
            self.air_id.as_bytes(),
            &self.generator_version.to_be_bytes(),
            self.parent_canonical_id.as_bytes(),
            self.stable_anchor_id.as_bytes(),
            &self.height.to_be_bytes(),
            self.miner_policy_digest.as_bytes(),
            self.payout_policy_digest.as_bytes(),
            &self.attempt_nonce.to_be_bytes(),
            self.body_core_root.as_bytes(),
            self.parent_derived_difficulty_context.as_bytes(),
            &self.time_bucket.to_be_bytes(),
        ];
        let mut out = Vec::new();
        for field in fields {
            out.extend_from_slice(&(field.len() as u64).to_be_bytes());
            out.extend_from_slice(field);
        }
        out
    }

    /// `seed = H_512(Enc(FW/INPUT/v2, x))` (eq. 5.2).
    pub fn seed(&self) -> DomainSeparatedHash {
        DomainSeparatedHash::new(labels::FW_INPUT, &[&self.encode()])
    }

    /// Expands the seed into the two operand tiles `X` and `Y` (eq. 5.3).
    pub fn instance(&self) -> FreshWorkInstance {
        let seed = self.seed();
        FreshWorkInstance {
            x: expand_seed(labels::FW_A, &seed),
            y: expand_seed(labels::FW_B, &seed),
            seed,
        }
    }
}

/// The operands generated from one Fresh Work seed.
#[derive(Clone, Debug)]
pub struct FreshWorkInstance {
    /// The seed both operands were expanded from.
    pub seed: DomainSeparatedHash,
    /// Left operand `X`.
    pub x: WorkUnit,
    /// Right operand `Y`.
    pub y: WorkUnit,
}

/// Expands a seed into a uniform INT8 tile by byte rejection (eq. 5.3):
/// `Map(b) = b - 127` for `b` in `{0, ..., 254}`, and `b = 255` is rejected,
/// so entries are exactly uniform on `[-127, 127]`.
///
/// The byte stream is the [`MockXof`] placeholder for SHAKE256.
pub fn expand_seed(label: &str, seed: &DomainSeparatedHash) -> WorkUnit {
    let mut xof = MockXof::new(label, seed);
    let mut data = Vec::with_capacity(TILE_ELEMS);
    while data.len() < TILE_ELEMS {
        let byte = xof.next_byte();
        if byte != u8::MAX {
            data.push((i16::from(byte) - 127) as i8);
        }
    }
    Matrix {
        data: data.into_boxed_slice(),
    }
}

/// Identifiers and lottery values of one evaluated attempt (eqs. 5.4–5.5).
#[derive(Clone, Copy, Debug, PartialEq, Eq)]

    pub output_root: DomainSeparatedHash,
    /// `WorkID = H_512(Enc(FW/ID/v2, x, outputRoot))`; also the ShareID.
    pub work_id: DomainSeparatedHash,
    /// `BlockID = H_512(Enc(BLOCK/v2, x, outputRoot))`.
    pub block_id: DomainSeparatedHash,
    /// Share lottery value `θ_S = H_512(Enc(FW/SHARE/v2, WorkID))`.
    pub share_ticket: DomainSeparatedHash,
    /// Seal lottery value `θ_B = H_512(Enc(FW/SEAL/v2, WorkID))`.
    pub seal_ticket: DomainSeparatedHash,
}

impl FreshWorkIds {
    /// Derives all identifiers from the context and the committed product.
    /// No witness material enters these maps (Theorem 5.1).
    pub fn derive(context: &FreshWorkContext, product: &AccumulationTarget) -> Self {
        let x = context.encode();
        let output_root = {
            let mut h = DomainHasher::new(labels::FW_OUTPUT);
            h.field_i32s(product.as_slice());
            h.finalize()
        };
        let work_id = DomainSeparatedHash::new(labels::FW_ID, &[&x, output_root.as_bytes()]);
        let block_id = DomainSeparatedHash::new(labels::BLOCK, &[&x, output_root.as_bytes()]);
        FreshWorkIds {
            output_root,
            work_id,
            block_id,
            share_ticket: DomainSeparatedHash::new(labels::FW_SHARE, &[work_id.as_bytes()]),
            seal_ticket: DomainSeparatedHash::new(labels::FW_SEAL, &[work_id.as_bytes()]),
        }
    }
}

// ---------------------------------------------------------------------------
// Freivalds verification over F_p (spec §3.2)
// ---------------------------------------------------------------------------

/// Checks `A(B v) = C v` over `F_p` for `rounds` independent challenge vectors.
///
/// The vectors are drawn only after `A`, `B` and `C` are bound into a
/// Fiat–Shamir transcript, with one domain per round and rejection sampling
/// to uniform `F_p` elements. Each round costs `3 n^2` field multiplications;
/// the normative six rounds cost `4,718,592 = (9/256) n^3`.
///
/// Returns `false` for `rounds == 0`. Consensus uses [`DEFAULT_FREIVALDS_ROUNDS`].
/// The matrix types already enforce the tile domains, which is what makes
/// agreement modulo `p` imply equality over the integers (eq. 3.3).
///
/// **PoC status:** the transcript hash and XOF are placeholders (see
/// [`primitives::DomainSeparatedHash`]); the field arithmetic is exact.
pub fn verify_freivalds_rounds(
    a: &Matrix<i8>,
    b: &Matrix<i8>,
    c: &Matrix<i32>,
    rounds: usize,
) -> bool {
    if rounds == 0 {
        return false;
    }
    let transcript = {
        let mut h = DomainHasher::new(labels::FW_FREIVALDS);
        h.field_i8s(a.as_slice())
            .field_i8s(b.as_slice())
            .field_i32s(c.as_slice());
        h.finalize()
    };
    (0..rounds as u64).all(|round| {
        let v = challenge_vector(&transcript, round);
        let bv = mat_vec(b, &v);
        mat_vec(a, &bv) == mat_vec(c, &v)
    })
}

/// Uniform challenge `v^(round)` in `F_p^n`: 61-bit draws, rejecting the single
/// value `p` so every element of `[0, p)` is equally likely.
fn challenge_vector(transcript: &DomainSeparatedHash, round: u64) -> Vec<Mersenne61> {
    let mut round_seed = DomainHasher::new(labels::FW_FREIVALDS);
    round_seed.field_digest(transcript).field_u64(round);
    let mut xof = MockXof::new(labels::FW_FREIVALDS, &round_seed.finalize());
    let mut v = Vec::with_capacity(TILE_DIM);
    while v.len() < TILE_DIM {
        let candidate = xof.next_u64() & Mersenne61::P;
        if candidate != Mersenne61::P {
            v.push(Mersenne61::new(candidate));
        }
    }
    v
}

fn mat_vec<T: TileElement>(m: &Matrix<T>, v: &[Mersenne61]) -> Vec<Mersenne61> {
    m.data
        .chunks_exact(TILE_DIM)
        .map(|row| signed_dot(row, v))
        .collect()
}

/// `sum_k row[k] * v[k]` over `F_p`. Positive and negative terms accumulate
/// exactly in `u128`: `|row[k]| <= 2^23` and `v[k] < 2^61`, so each sum stays
/// below `512 * 2^84 = 2^93`.
fn signed_dot<T: TileElement>(row: &[T], v: &[Mersenne61]) -> Mersenne61 {
    let (mut positive, mut negative) = (0u128, 0u128);
    for (&x, y) in row.iter().zip(v) {
        let x: i64 = x.into();
        let term = u128::from(x.unsigned_abs()) * u128::from(y.value());
        if x >= 0 {
            positive += term;
        } else {
            negative += term;
        }
    }
    Mersenne61::new(Mersenne61::reduce_u128(positive))
        - Mersenne61::new(Mersenne61::reduce_u128(negative))
}

//! Module 1 (SciVM) — the Ozaki-II numerical profile over native INT8 tiles.
//!
//! The Scientific Virtual Machine executes admitted commercial and scientific
//! tasks with **zero consensus weight** (spec §1.1, eq. 3.23). Dense
//! high-precision matrix products are carried out by deterministic Ozaki-II
//! residue splitting (spec §3.6): scaled integer operands are split into
//! centered residues modulo `r` pairwise-coprime auxiliary moduli
//! `2 <= q_j <= 255`, multiplied as exact INT8 x INT8 -> INT32 tiles, reduced
//! modulo each `q_j`, and recombined by centered CRT. The base field `F_p` and
//! Module 3 are unchanged; auxiliary moduli never replace `p`.
//!
//! **PoC status:** the profile, residue map, accumulation and CRT are exact
//! integer code. The STARK `TraceCheck`, `Scale`/`Unscale` of floating-point
//! formats and the remaining `Sem` semantics are out of scope.

use core::fmt;

use work::{ACCUMULATION_BOUND, INT8_MAX, INT8_MIN, TILE_DIM};

/// Smallest admissible auxiliary modulus.
pub const MIN_MODULUS: u32 = 2;
/// Largest admissible auxiliary modulus; centered residues then fit `[-127, 127]`.
pub const MAX_MODULUS: u32 = 255;
/// Contribution of any SciVM task to chain weight, `ΔW_consensus(τ) = 0`.
pub const CONSENSUS_WEIGHT: u64 = 0;
/// Largest magnitude at which every integer is exactly representable in `f64`.
pub const MAX_EXACT_F64_INTEGER: f64 = 9_007_199_254_740_992.0; // 2^53
/// PoC limit: centered CRT is evaluated in `i128`, so `Q` must stay below `2^127`.
pub const MAX_RECONSTRUCTION_MODULUS: u128 = 1 << 127;

/// Rounding rule `Rnd` of the target format.
#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, Hash)]
pub enum RoundingMode {
    /// IEEE 754 round to nearest, ties to even.
    #[default]
    NearestTiesToEven,
    /// Round toward zero.
    TowardZero,
    /// Round toward positive infinity.
    TowardPositive,
    /// Round toward negative infinity.
    TowardNegative,
}

/// The task-bound numerical profile `Π_num` (eq. 3.20), restricted to the
/// fields that govern Ozaki-II residue splitting and exact reconstruction.
///
/// The profile is immutably bound by the client at admission; two executions
/// with the same program, inputs, state and profile produce identical outputs.
#[derive(Clone, Debug, PartialEq, Eq, Hash)]
pub struct OzakiIIProfile {
    /// Profile version `ν`, which also fixes the Ozaki-II construction.
    pub version: u32,
    /// Auxiliary moduli `q = (q_1, ..., q_r)`: pairwise coprime, each in `[2, 255]`.
    pub moduli: Vec<u32>,
    /// Common inner dimension `K` of the operands.
    pub inner_dim: u64,
    /// Certified bound `B_X >= max |X~|` on the scaled left operand.
    pub bound_x: u64,
    /// Certified bound `B_Y >= max |Y~|` on the scaled right operand.
    pub bound_y: u64,
    /// Rounding rule of the output format.
    pub rounding: RoundingMode,
}

/// Rejection reasons of the SciVM profile and residue pipeline.
#[derive(Clone, Debug, PartialEq)]
pub enum SciVmError {
    /// `r = 0`: at least one auxiliary modulus is required.
    NoModuli,
    /// A modulus lies outside `[2, 255]`.
    ModulusOutOfRange(u32),
    /// Two moduli share a factor.
    ModuliNotCoprime(u32, u32),
    /// `K = 0`.
    ZeroInnerDimension,
    /// An operand is not a finite integer of magnitude at most `2^53`.
    NonIntegralOperand {
        /// Position in the input slice.
        index: usize,
        /// The offending value.
        value: f64,
    },
    /// An operand exceeds its certified bound `B_X` or `B_Y`.
    OperandOutOfBound {
        /// Position in the input slice.
        index: usize,
        /// The offending magnitude.
        magnitude: u64,
        /// The certified bound.
        bound: u64,
    },
    /// `Q <= 2 B_rec`: the moduli cannot reconstruct the product exactly.
    InsufficientReconstructionCapacity {
        /// `Q = prod q_j`.
        modulus: u128,
        /// `2 B_rec`.
        twice_bound: u128,
    },
    /// `Q` or `B_rec` exceeds the PoC's `i128` reconstruction range.
    ReconstructionRangeExceeded,
    /// Operand slices or residue vectors have inconsistent lengths.
    LengthMismatch,
}

impl fmt::Display for SciVmError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            SciVmError::NoModuli => write!(f, "profile needs at least one modulus"),
            SciVmError::ModulusOutOfRange(q) => {
                write!(f, "modulus {q} outside [{MIN_MODULUS}, {MAX_MODULUS}]")
            }
            SciVmError::ModuliNotCoprime(a, b) => write!(f, "moduli {a} and {b} are not coprime"),
            SciVmError::ZeroInnerDimension => write!(f, "inner dimension K must be positive"),
            SciVmError::NonIntegralOperand { index, value } => {
                write!(f, "operand {index} = {value} is not an exact integer")
            }
            SciVmError::OperandOutOfBound {
                index,
                magnitude,
                bound,
            } => {
                write!(
                    f,
                    "operand {index} has |x| = {magnitude} > certified bound {bound}"
                )
            }
            SciVmError::InsufficientReconstructionCapacity {
                modulus,
                twice_bound,
            } => {
                write!(f, "Q = {modulus} <= 2 B_rec = {twice_bound}")
            }
            SciVmError::ReconstructionRangeExceeded => {
                write!(f, "Q or B_rec exceeds the PoC i128 reconstruction range")
            }
            SciVmError::LengthMismatch => write!(f, "inconsistent operand or residue lengths"),
        }
    }
}

impl std::error::Error for SciVmError {}

impl OzakiIIProfile {
    /// Checks the admission conditions of eq. (3.22): `r >= 1`, every
    /// `2 <= q_j <= 255`, pairwise coprimality, and `Q > 2 B_rec` with
    /// `B_rec = K B_X B_Y`. Insufficient capacity is a rejection, never a
    /// silent approximation.
    pub fn validate(&self) -> Result<(), SciVmError> {
        validate_moduli(&self.moduli)?;
        if self.inner_dim == 0 {
            return Err(SciVmError::ZeroInnerDimension);
        }
        let modulus = self.reconstruction_modulus()?;
        let twice_bound = self
            .reconstruction_bound()?
            .checked_mul(2)
            .ok_or(SciVmError::ReconstructionRangeExceeded)?;
        if modulus <= twice_bound {
            return Err(SciVmError::InsufficientReconstructionCapacity {
                modulus,
                twice_bound,
            });
        }
        Ok(())
    }

    /// `Q = prod q_j`, which must stay below [`MAX_RECONSTRUCTION_MODULUS`] in the PoC.
    pub fn reconstruction_modulus(&self) -> Result<u128, SciVmError> {
        reconstruction_modulus(&self.moduli)
    }

    /// `B_rec = K B_X B_Y`, the bound on every entry of the scaled integer product.
    pub fn reconstruction_bound(&self) -> Result<u128, SciVmError> {
        u128::from(self.inner_dim)
            .checked_mul(u128::from(self.bound_x))
            .and_then(|v| v.checked_mul(u128::from(self.bound_y)))
            .ok_or(SciVmError::ReconstructionRangeExceeded)
    }

    /// Exact dot product `sum_k x_k y_k` of scaled integer operands through the
    /// Ozaki-II path: residue split, INT8 x INT8 -> INT32 accumulation per
    /// 512-wide tile, reduction modulo each `q_j`, then centered CRT.
    ///
    /// This is one entry of the tile product of eq. (3.21); a full matrix
    /// product repeats it for every output entry.
    pub fn exact_dot(&self, x: &[f64], y: &[f64]) -> Result<i128, SciVmError> {
        self.validate()?;
        if x.len() != y.len() || x.len() as u64 != self.inner_dim {
            return Err(SciVmError::LengthMismatch);
        }
        check_bound(x, self.bound_x)?;
        check_bound(y, self.bound_y)?;
        let k = x.len();
        let rx = try_split_residues(x, &self.moduli)?;
        let ry = try_split_residues(y, &self.moduli)?;

        let mut assembled = Vec::with_capacity(self.moduli.len());
        for (j, &q) in self.moduli.iter().enumerate() {
            let (xj, yj) = (&rx[j * k..(j + 1) * k], &ry[j * k..(j + 1) * k]);
            let mut acc_mod_q: i64 = 0;
            for (tile_x, tile_y) in xj.chunks(TILE_DIM).zip(yj.chunks(TILE_DIM)) {
                // One native tile: at most 512 INT8 products, so the INT32
                // accumulator stays within ±8,258,048 (eq. 3.22).
                let tile: i32 = tile_x
                    .iter()
                    .zip(tile_y)
                    .map(|(&a, &b)| i32::from(a) * i32::from(b))
                    .sum();
                debug_assert!(tile.abs() <= ACCUMULATION_BOUND);
                acc_mod_q = (acc_mod_q + i64::from(tile)).rem_euclid(i64::from(q));
            }
            assembled.push(acc_mod_q);
        }
        crt_centered(&assembled, &self.moduli)
    }
}

/// Splits scaled integer operands into centered residues
/// `cres_q(z) = z - q * floor((z + floor(q/2)) / q)` for every modulus.
///
/// The moduli must be **pairwise-coprime integers in `[2, 255]`**, so that each
/// residue lies in `[-127, 127]`: exactly the native INT8 tile domain, with
/// `-128` never produced. Operands must be finite integers with `|z| <= 2^53`
/// (the output of the profile's `Scale` step).
///
/// The output is residue-major: entry `j * n + i` is `cres_{q_j}(z_i)` for
/// `n = scaled_operands.len()`.
///
/// # Panics
///
/// Panics on invalid moduli or operands; use [`try_split_residues`] to handle
/// these as errors.
pub fn split_residues(scaled_operands: &[f64], moduli: &[u32]) -> Vec<i8> {
    try_split_residues(scaled_operands, moduli).unwrap_or_else(|e| panic!("split_residues: {e}"))
}

/// Fallible form of [`split_residues`].
pub fn try_split_residues(scaled_operands: &[f64], moduli: &[u32]) -> Result<Vec<i8>, SciVmError> {
    validate_moduli(moduli)?;
    let integers = scaled_operands
        .iter()
        .enumerate()
        .map(|(index, &value)| {
            exact_integer(value).ok_or(SciVmError::NonIntegralOperand { index, value })
        })
        .collect::<Result<Vec<i64>, _>>()?;
    let mut out = Vec::with_capacity(moduli.len() * integers.len());
    for &q in moduli {
        out.extend(integers.iter().map(|&z| centered_residue(z, q)));
    }
    Ok(out)
}

/// `cres_q(z)` for one value; the result lies in `[-floor(q/2), q - 1 - floor(q/2)]`.
pub fn centered_residue(z: i64, q: u32) -> i8 {
    let q = i64::from(q);
    let residue = z - q * (z + q / 2).div_euclid(q);
    debug_assert!((i64::from(INT8_MIN)..=i64::from(INT8_MAX)).contains(&residue));
    residue as i8
}

/// Centered CRT reconstruction `CRT_0`: the unique `Z` in
/// `[-floor(Q/2), ceil(Q/2) - 1]` with `Z = r_j (mod q_j)` for every `j`.
pub fn crt_centered(residues: &[i64], moduli: &[u32]) -> Result<i128, SciVmError> {
    validate_moduli(moduli)?;
    if residues.len() != moduli.len() {
        return Err(SciVmError::LengthMismatch);
    }
    let modulus = reconstruction_modulus(moduli)?;
    // Garner-style incremental combination; every intermediate is < Q < 2^127.
    let (mut value, mut combined) = (0u128, 1u128);
    for (&r, &q) in residues.iter().zip(moduli) {
        let q128 = u128::from(q);
        let target = r.rem_euclid(i64::from(q)) as u128;
        let current = value % q128;
        let diff = (target + q128 - current) % q128;
        let inverse = mod_inverse((combined % q128) as u32, q) as u128;
        let step = diff * inverse % q128;
        value += combined * step;
        combined *= q128;
    }
    debug_assert_eq!(combined, modulus);
    let half_up = modulus - modulus / 2; // ceil(Q/2)
    Ok(if value >= half_up {
        value as i128 - modulus as i128
    } else {
        value as i128
    })
}

fn validate_moduli(moduli: &[u32]) -> Result<(), SciVmError> {
    if moduli.is_empty() {
        return Err(SciVmError::NoModuli);
    }
    if let Some(&q) = moduli
        .iter()
        .find(|q| !(MIN_MODULUS..=MAX_MODULUS).contains(q))
    {
        return Err(SciVmError::ModulusOutOfRange(q));
    }
    for (i, &a) in moduli.iter().enumerate() {
        if let Some(&b) = moduli[i + 1..].iter().find(|&&b| gcd(a, b) != 1) {
            return Err(SciVmError::ModuliNotCoprime(a, b));
        }
    }
    Ok(())
}

fn reconstruction_modulus(moduli: &[u32]) -> Result<u128, SciVmError> {
    moduli
        .iter()
        .try_fold(1u128, |acc, &q| acc.checked_mul(u128::from(q)))
        .filter(|&m| m < MAX_RECONSTRUCTION_MODULUS)
        .ok_or(SciVmError::ReconstructionRangeExceeded)
}

fn check_bound(values: &[f64], bound: u64) -> Result<(), SciVmError> {
    for (index, &value) in values.iter().enumerate() {
        let magnitude = exact_integer(value)
            .ok_or(SciVmError::NonIntegralOperand { index, value })?
            .unsigned_abs();
        if magnitude > bound {
            return Err(SciVmError::OperandOutOfBound {
                index,
                magnitude,
                bound,
            });
        }
    }
    Ok(())
}

fn exact_integer(value: f64) -> Option<i64> {
    (value.is_finite() && value.fract() == 0.0 && value.abs() <= MAX_EXACT_F64_INTEGER)
        .then_some(value as i64)
}

fn gcd(mut a: u32, mut b: u32) -> u32 {
    while b != 0 {
        (a, b) = (b, a % b);
    }
    a
}

/// Inverse of `a` modulo `q`; callers guarantee `gcd(a, q) = 1`.
fn mod_inverse(a: u32, q: u32) -> u32 {
    let (mut old_r, mut r) = (i64::from(a), i64::from(q));
    let (mut old_s, mut s) = (1i64, 0i64);
    while r != 0 {
        let quotient = old_r / r;
        (old_r, r) = (r, old_r - quotient * r);
        (old_s, s) = (s, old_s - quotient * s);
    }
    debug_assert_eq!(old_r, 1, "modulus not coprime");
    old_s.rem_euclid(i64::from(q)) as u32
}

//! Module 4 — deterministic scheduler and sequential D'Hondt allocation.
//!
//! The unit of execution entitlement is a discrete, immutable ShareID of
//! verified mature work — never an address (spec §6.1). Each slot goes to the
//! eligible ShareID with the largest D'Hondt quotient `w_s / (a_s + 1)`, ties
//! broken by the smallest ShareID (eq. 6.4). Quotients are compared exactly by
//! cross-multiplication (eq. 6.5); no floating point is involved.
//!
//! Owner labels enter only through bonded capacity `K_i`. Without binding
//! capacity, an entity's allocation is the sum over its ShareIDs and does not
//! depend on how they are partitioned into owners or addresses: there is no
//! aggregation premium (Theorem 6.3).
//!
//! **PoC status:** the normative weight `w_s = floor(2^640 / Θ_S(s))` (eq. 6.1)
//! needs a 640-bit integer. The PoC narrows [`ShareWeight`] to `u64`, which
//! keeps every cross-product `w_s (a_r + 1)` exact in `u128`.

use core::fmt;
use std::collections::{BTreeMap, BTreeSet};

use primitives::DomainSeparatedHash;
pub use primitives::ShareWeight;

/// Beneficiary bound in the Fresh Work context; enters D'Hondt only via capacity.
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct OwnerId(pub u64);

/// A discrete unit of execution entitlement: one verified mature Fresh Work share.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
pub struct ShareID {
    /// `ShareID = WorkID`, a 64-byte digest compared as a canonical string.
    pub id: DomainSeparatedHash,
    /// Integer weight fixed by the consensus target of the parent block.
    pub weight: ShareWeight,
    /// Owner, used only to look up bonded lease capacity.
    pub owner: OwnerId,
}

/// Concurrent-lease capacity `K_i^free` per owner for one batch.
#[derive(Clone, Debug, Default, PartialEq, Eq)]
pub enum Capacity {
    /// No binding capacity: every ShareID stays eligible.
    #[default]
    Unbounded,
    /// Free capacity per owner; owners absent from the map have none.
    PerOwner(BTreeMap<OwnerId, u64>),
}

/// Rejection reasons when forming a cohort.
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SchedulerError {
    /// The same ShareID appears twice (identifiers are unique, eq. 7.23).
    DuplicateShareId(DomainSeparatedHash),
    /// A ShareID has zero weight; normative weights are at least `2^128`.
    ZeroWeight(DomainSeparatedHash),
}

impl fmt::Display for SchedulerError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            SchedulerError::DuplicateShareId(id) => write!(f, "duplicate ShareID {id:?}"),
            SchedulerError::ZeroWeight(id) => write!(f, "ShareID {id:?} has zero weight"),
        }
    }
}

impl std::error::Error for SchedulerError {}

/// Result of one allocation batch.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Allocation {
    /// `slots[t]` is the index (into the cohort's shares) of the ShareID that
    /// receives slot `t`, in canonical lease order.
    pub slots: Vec<usize>,
    /// Slots each ShareID received in this batch, indexed like the shares.
    pub per_share: Vec<u64>,
}

impl Allocation {
    /// Number of slots assigned (less than requested only if capacity ran out).
    pub fn assigned(&self) -> usize {
        self.slots.len()
    }
}

/// A snapshot `S` of mature qualifying ShareIDs with persistent counters `a_s`.
///
/// Counters persist across the batches of a cohort, so allocating `N_1` then
/// `N_2` slots without binding capacity equals allocating `N_1 + N_2` at once.
#[derive(Clone, Debug)]
pub struct Cohort {
    shares: Vec<ShareID>,
    counters: Vec<u64>,
}

impl Cohort {
    /// Forms a cohort, rejecting duplicate ShareIDs and zero weights.
    pub fn new(shares: Vec<ShareID>) -> Result<Self, SchedulerError> {
        let mut seen = BTreeSet::new();
        for share in &shares {
            if share.weight == 0 {
                return Err(SchedulerError::ZeroWeight(share.id));
            }
            if !seen.insert(share.id) {
                return Err(SchedulerError::DuplicateShareId(share.id));
            }
        }
        let counters = vec![0; shares.len()];
        Ok(Cohort { shares, counters })
    }

    /// The cohort's ShareIDs, in input order.
    pub fn shares(&self) -> &[ShareID] {
        &self.shares
    }

    /// Cumulative slots `a_s` held by each ShareID in this cohort.
    pub fn counters(&self) -> &[u64] {
        &self.counters
    }

    /// Total weight `W = sum w_s`.
    pub fn total_weight(&self) -> u128 {
        self.shares.iter().map(|s| u128::from(s.weight)).sum()
    }

    /// Runs the sequential D'Hondt recursion (eq. 6.4) for up to `slots` steps.
    ///
    /// Each step assigns exactly one slot to `argmax w_s / (a_s + 1)` over the
    /// eligible set `E_t = {s : K_owner(s) > 0}`, ties to the least ShareID,
    /// then increments that ShareID's counter and decrements its owner's
    /// capacity. The loop stops when `t = N` or `E_t` is empty.
    pub fn allocate_batch(&mut self, slots: usize, capacity: &Capacity) -> Allocation {
        let mut free: Option<BTreeMap<OwnerId, u64>> = match capacity {
            Capacity::Unbounded => None,
            Capacity::PerOwner(map) => Some(map.clone()),
        };
        let mut allocation = Allocation {
            slots: Vec::with_capacity(slots),
            per_share: vec![0; self.shares.len()],
        };
        while allocation.slots.len() < slots {
            let eligible = |s: &ShareID| match &free {
                None => true,
                Some(map) => map.get(&s.owner).is_some_and(|&k| k > 0),
            };
            let Some(winner) = self.select(eligible) else {
                break;
            };
            self.counters[winner] += 1;
            allocation.per_share[winner] += 1;
            allocation.slots.push(winner);
            if let Some(map) = free.as_mut()
                && let Some(k) = map.get_mut(&self.shares[winner].owner)
            {
                *k -= 1;
            }
        }
        allocation
    }

    /// Index of the eligible ShareID with the largest quotient, ties to the least ID.
    fn select(&self, eligible: impl Fn(&ShareID) -> bool) -> Option<usize> {
        let mut best: Option<usize> = None;
        for (s, share) in self.shares.iter().enumerate() {
            if !eligible(share) {
                continue;
            }
            best = match best {
                None => Some(s),
                Some(b) => match self.compare_quotients(s, b) {
                    core::cmp::Ordering::Greater => Some(s),
                    core::cmp::Ordering::Less => Some(b),
                    core::cmp::Ordering::Equal if share.id < self.shares[b].id => Some(s),
                    core::cmp::Ordering::Equal => Some(b),
                },
            };
        }
        best
    }

    /// Exact comparison `w_s/(a_s+1)` vs `w_r/(a_r+1)` as `w_s (a_r+1)` vs `w_r (a_s+1)`.
    fn compare_quotients(&self, s: usize, r: usize) -> core::cmp::Ordering {
        let lhs = u128::from(self.shares[s].weight) * (u128::from(self.counters[r]) + 1);
        let rhs = u128::from(self.shares[r].weight) * (u128::from(self.counters[s]) + 1);
        lhs.cmp(&rhs)
    }
}

/// Allocates `available_slots` to discrete ShareIDs by sequential D'Hondt,
/// for a fresh cohort without binding capacity.
///
/// Quotients are `w_s / (a_s + 1)` and slots are assigned strictly to
/// individual ShareIDs — never to owners — so splitting or merging holdings
/// across addresses cannot change any entity's total (no aggregation premium).
/// Every slot is assigned (Theorem 6.1).
pub fn dhondt_allocation(
    shares: &[ShareID],
    available_slots: usize,
) -> Result<Allocation, SchedulerError> {
    Ok(Cohort::new(shares.to_vec())?.allocate_batch(available_slots, &Capacity::Unbounded))
}

//! Module 5 — hybrid state, supply invariant and tokenomics.
//!
//! The supply is fixed at `M = 1.5 * 10^15` atoms and is partitioned into
//! eight disjoint, non-negative balance categories that sum to `M` in every
//! reachable state (eq. 7.19). Every operation moves value between categories
//! by deltas that sum to zero (eq. 7.13) and commits atomically or not at all
//! (eq. 7.14).
//!
//! Two sub-ledgers refine categories without adding a balance:
//!
//! * **FounderAllocation** — 1,460 time-locked Genesis UTXOs inside
//!   [`StateCategory::Circulating`] (`U`). Vesting changes only spendability;
//!   it "adds no ninth balance" (§7.6).
//! * **MatureFeeLots** — independent, immutable fee lots inside
//!   [`StateCategory::EmissionReserve`] (`R_a`), alongside the Genesis emission
//!   lot and the permanently segregated rounding dust (§7.8).
//!
//! **PoC status:** `U`, `S_z`, `B_bond` and `V_L2` are aggregate balances
//! (no per-UTXO ownership, notes or signatures). Epoch length, maturation
//! delay `L`, emission rate `η` and fee tariffs are deployment parameters;
//! [`LedgerParams::poc`] picks illustrative values.

use core::fmt;
use std::collections::BTreeMap;

use primitives::{Atoms, ShareWeight, TOTAL_SUPPLY};

/// Genesis Founder Allocation `F = M / 10 = 150,000,000,000,000` atoms.
pub const FOUNDER_ALLOCATION: Atoms = Atoms(150_000_000_000_000);
/// Genesis Network Emission Reserve `R_a^(0) = M - F = 1,350,000,000,000,000` atoms.
pub const GENESIS_EMISSION_RESERVE: Atoms = Atoms(1_350_000_000_000_000);
/// Number of daily founder vesting tranches, `D_vest`.
pub const VESTING_TRANCHES: u64 = 1_460;
/// Nominal block interval `T_B` in seconds.
pub const BLOCK_INTERVAL_SECONDS: u64 = 120;
/// Blocks per nominal day, `B_day = 86,400 s / T_B`.
pub const BLOCKS_PER_DAY: u64 = 720;
/// Blocks until full vesting, `D_vest * B_day` (four 365-day years).
pub const VESTING_BLOCKS: u64 = 1_051_200;

const _: () = assert!(FOUNDER_ALLOCATION.0 * 10 == TOTAL_SUPPLY.0);
const _: () = assert!(FOUNDER_ALLOCATION.0 + GENESIS_EMISSION_RESERVE.0 == TOTAL_SUPPLY.0);
const _: () = assert!(86_400 / BLOCK_INTERVAL_SECONDS == BLOCKS_PER_DAY);
const _: () = assert!(VESTING_TRANCHES == 4 * 365);
const _: () = assert!(VESTING_TRANCHES * BLOCKS_PER_DAY == VESTING_BLOCKS);

/// The eight disjoint balance categories of eq. (7.19).
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub enum StateCategory {
    /// `U` — public UTXOs, including the FounderAllocation tranches.
    Circulating,
    /// `S_z` — shielded pool balance (equals the sum of unspent notes).
    ShieldedPool,
    /// `E_esc` — job escrows.
    Escrow,
    /// `B_bond` — executor bonds.
    Bonds,
    /// `V_L2` — L2 settlement-slot vaults.
    L2Vaults,
    /// `R_a` — available reserve: Genesis emission lot, MatureFeeLots and segregated dust.
    EmissionReserve,
    /// `R_l` — locked reserve: Base Fees and queue fees awaiting maturation.
    LockedBaseFees,
    /// `P_pending` — booked Fresh Work rewards not yet paid.
    PendingRewards,
}

impl StateCategory {
    /// Number of categories.
    pub const COUNT: usize = 8;

    /// All categories in canonical order.
    pub const ALL: [StateCategory; StateCategory::COUNT] = [
        StateCategory::Circulating,
        StateCategory::ShieldedPool,
        StateCategory::Escrow,
        StateCategory::Bonds,
        StateCategory::L2Vaults,
        StateCategory::EmissionReserve,
        StateCategory::LockedBaseFees,
        StateCategory::PendingRewards,
    ];

    /// Position in [`StateCategory::ALL`].
    pub const fn index(self) -> usize {
        self as usize
    }

    /// The symbol used in the specification.
    pub const fn symbol(self) -> &'static str {
        match self {
            StateCategory::Circulating => "U",
            StateCategory::ShieldedPool => "S_z",
            StateCategory::Escrow => "E_esc",
            StateCategory::Bonds => "B_bond",
            StateCategory::L2Vaults => "V_L2",
            StateCategory::EmissionReserve => "R_a",
            StateCategory::LockedBaseFees => "R_l",
            StateCategory::PendingRewards => "P_pending",
        }
    }
}

/// Signed per-category change produced by one operation.
#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
pub struct BalanceDelta([i128; StateCategory::COUNT]);

impl BalanceDelta {
    /// The empty delta.
    pub const ZERO: BalanceDelta = BalanceDelta([0; StateCategory::COUNT]);

    /// Moves `amount` from one category to another.
    pub fn transfer(from: StateCategory, to: StateCategory, amount: Atoms) -> Self {
        Self::ZERO
            .with(from, -i128::from(amount.get()))
            .with(to, i128::from(amount.get()))
    }

    /// Adds `change` to one category's entry.
    pub fn with(mut self, category: StateCategory, change: i128) -> Self {
        self.0[category.index()] += change;
        self
    }

    /// Change for one category.
    pub fn get(&self, category: StateCategory) -> i128 {
        self.0[category.index()]
    }

    /// Sum over all categories; zero for every admissible delta (eq. 7.13).
    pub fn sum(&self) -> i128 {
        self.0.iter().sum()
    }
}

/// Exact emission rate `η = u / v` with `0 < u < v` and `gcd(u, v) = 1`.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
pub struct EmissionRate {
    numerator: u64,
    denominator: u64,
}

impl EmissionRate {
    /// Validates and builds `η = u / v`.
    pub fn new(numerator: u64, denominator: u64) -> Result<Self, LedgerError> {
        if numerator == 0 || numerator >= denominator || gcd(numerator, denominator) != 1 {
            return Err(LedgerError::InvalidEmissionRate {
                numerator,
                denominator,
            });
        }
        Ok(EmissionRate {
            numerator,
            denominator,
        })
    }

    /// `u`.
    pub fn numerator(&self) -> u64 {
        self.numerator
    }

    /// `v`.
    pub fn denominator(&self) -> u64 {
        self.denominator
    }

    /// Exact integer ceiling budget `ceil(u A / v) = floor((u A + v - 1) / v)` (eq. 7.25).
    /// Since `u < v`, the budget never exceeds the principal `A`.
    pub fn budget(&self, principal: Atoms) -> Atoms {
        let (u, v) = (u128::from(self.numerator), u128::from(self.denominator));
        Atoms((u * u128::from(principal.get())).div_ceil(v) as u64)
    }
}

    /// Maturation delay `L` in epochs: a fee paid in epoch `e` matures in `e + L + 1`.
    pub maturation_delay: u64,
    /// Emission rate `η` applied to every eligible lot each epoch.

/// One time-locked Genesis founder UTXO `g_d`.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct FounderTranche {
    /// Tranche index `d` in `1..=1460`.
    pub day: u64,
    /// Immutable value `a_d = V_d - V_{d-1}`.
    pub value: Atoms,
    /// Unlock height `h_d = h_G + d * B_day`.
    pub unlock_height: u64,
}

/// The FounderAllocation: 1,460 daily UTXO tranches over four nominal years.
///
/// `V_d = floor(F d / D_vest)` and `a_d = V_d - V_{d-1}`; the telescoping
/// differences allocate every atom of `F` exactly once with no rounding loss.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct FounderVesting {
    genesis_height: u64,
    allocation: Atoms,
}

impl FounderVesting {
    /// The vesting schedule of `allocation` starting at `genesis_height`.
    pub const fn new(genesis_height: u64, allocation: Atoms) -> Self {
        FounderVesting {
            genesis_height,
            allocation,
        }
    }

    /// Total allocation `F`.
    pub fn allocation(&self) -> Atoms {
        self.allocation
    }

    /// Cumulative eligible atoms after day `d`, `V_d` (clamped at `D_vest`).
    pub fn cumulative(&self, day: u64) -> Atoms {
        let d = u128::from(day.min(VESTING_TRANCHES));
        Atoms((u128::from(self.allocation.get()) * d / u128::from(VESTING_TRANCHES)) as u64)
    }

    /// Tranche `g_d` for `1 <= d <= 1460`.
    pub fn tranche(&self, day: u64) -> Option<FounderTranche> {
        (1..=VESTING_TRANCHES)
            .contains(&day)
            .then(|| FounderTranche {
                day,
                value: Atoms(self.cumulative(day).get() - self.cumulative(day - 1).get()),
                unlock_height: self.genesis_height + day * BLOCKS_PER_DAY,
            })
    }

    /// All 1,460 tranches in order.
    pub fn tranches(&self) -> impl Iterator<Item = FounderTranche> + '_ {
        (1..=VESTING_TRANCHES).filter_map(|d| self.tranche(d))
    }

    /// Completed vesting intervals `d(h) = min(D_vest, max(0, floor((h - h_G) / B_day)))`.
    pub fn completed_intervals(&self, height: u64) -> u64 {
        (height.saturating_sub(self.genesis_height) / BLOCKS_PER_DAY).min(VESTING_TRANCHES)
    }

    /// Cumulative unlocked value `V_F(h) = V_{d(h)}`.
    pub fn vested_at(&self, height: u64) -> Atoms {
        self.cumulative(self.completed_intervals(height))
    }

    /// Value still time-locked at `height`, `F - V_F(h)`.
    pub fn locked_at(&self, height: u64) -> Atoms {
        Atoms(self.allocation.get() - self.vested_at(height).get())
    }

    /// `FounderOK` without the UTXO-presence conjunct: the height lock has
    /// passed and the spend carries the treasury dual signature (eq. 2.5).
    pub fn can_spend(&self, day: u64, height: u64, treasury_authorized: bool) -> bool {
        self.tranche(day)
            .is_some_and(|t| height >= t.unlock_height && treasury_authorized)
    }
}

/// Unique identifier of a reserve lot; lot `0` is the Genesis emission lot `c_G`.
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub struct LotId(pub u64);

/// An independent, immutable reserve lot `c`.
///
/// Each epoch the lot budgets `b = ceil(η A)` of its unbudgeted principal `A`
/// and pays every mature share `floor(b w_s / W_e)`. The flooring remainder
/// `r = b - sum floor(...)` is dust: it leaves the lot's principal and is
/// permanently segregated in `R_a`, never funding a later budget. Lots are
/// never merged, topped up or matured twice, so for every lot
/// `source_fee = principal + booked_total + dust_total` at all times.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct FeeLot {
    id: LotId,
    source_fee: Atoms,
    paid_epoch: u64,
    maturity_epoch: u64,
    principal: Atoms,
    booked_total: Atoms,
    dust_total: Atoms,
}

/// One lot's contribution to one epoch.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct LotEpochPayout {
    /// The lot.
    pub lot: LotId,
    /// Ceiling budget `b_{c,e}`.
    pub budget: Atoms,
    /// `Reward_{s,c,e}`, indexed like the input weights.
    pub rewards: Vec<Atoms>,
    /// Booked payouts `p_{c,e} = sum_s Reward_{s,c,e}`.
    pub booked: Atoms,
    /// Unpayable dust `r_{c,e} = b_{c,e} - p_{c,e}`.
    pub dust: Atoms,
}

impl FeeLot {
    /// The Genesis emission lot `c_G` with `A_{c_G,0} = R_0`.
    pub fn genesis(reserve: Atoms) -> Self {
        FeeLot {
            id: LotId(0),
            source_fee: reserve,
            paid_epoch: 0,
            maturity_epoch: 0,
            principal: reserve,
            booked_total: Atoms::ZERO,
            dust_total: Atoms::ZERO,
        }
    }

    /// A fresh lot for a matured fee `F_c` paid in `paid_epoch`.
    pub fn matured(id: LotId, fee: Atoms, paid_epoch: u64, maturity_epoch: u64) -> Self {
        FeeLot {
            id,
            source_fee: fee,
            paid_epoch,
            maturity_epoch,
            principal: fee,
            booked_total: Atoms::ZERO,
            dust_total: Atoms::ZERO,
        }
    }

    /// Lot identifier.
    pub fn id(&self) -> LotId {
        self.id
    }
    /// Original fee `F_c` (or `R_0` for the Genesis lot).
    pub fn source_fee(&self) -> Atoms {
        self.source_fee
    }
    /// Epoch `e_c` in which the fee was paid.
    pub fn paid_epoch(&self) -> u64 {
        self.paid_epoch
    }
    /// First eligible epoch `m_c = e_c + L + 1`.
    pub fn maturity_epoch(&self) -> u64 {
        self.maturity_epoch
    }
    /// Unbudgeted principal `A_{c,e}`.
    pub fn principal(&self) -> Atoms {
        self.principal
    }
    /// Cumulative booked payouts.
    pub fn booked_total(&self) -> Atoms {
        self.booked_total
    }
    /// Cumulative segregated dust.
    pub fn dust_total(&self) -> Atoms {
        self.dust_total
    }

    /// Settles one epoch against the mature share weights `w_s` (eq. 7.25).
    ///
    /// With `W_e = 0` the budget and every payout are zero. All divisions are
    /// exact Euclidean integer divisions; `0 <= p <= b <= A` always holds.
    pub fn settle_epoch(&mut self, rate: &EmissionRate, weights: &[ShareWeight]) -> LotEpochPayout {
        let total_weight: u128 = weights.iter().map(|&w| u128::from(w)).sum();
        let budget = if total_weight == 0 {
            Atoms::ZERO
        } else {
            rate.budget(self.principal)
        };
        let rewards: Vec<Atoms> = weights
            .iter()
            .map(|&w| {
                let reward = (u128::from(budget.get()) * u128::from(w)).checked_div(total_weight);
                Atoms(reward.unwrap_or(0) as u64)
            })
            .collect();
        let booked = Atoms(rewards.iter().map(|r| r.get()).sum());
        let dust = Atoms(budget.get() - booked.get());
        self.principal = Atoms(self.principal.get() - budget.get());
        self.booked_total = Atoms(self.booked_total.get() + booked.get());
        self.dust_total = Atoms(self.dust_total.get() + dust.get());
        LotEpochPayout {
            lot: self.id,
            budget,
            rewards,
            booked,
            dust,
        }
    }
}

/// Identifier of an escrowed job.
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]

/// Rejection reasons. A rejected operation leaves the state unchanged.
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum LedgerError {
    /// The delta does not sum to zero (eq. 7.13).
    NonZeroSumDelta(i128),
    /// A category would become negative.
    NegativeBalance(StateCategory),
    /// A category would exceed `u64`.
    BalanceOverflow(StateCategory),
    /// The categories would not sum to `M`.
    SupplyViolation {
        /// The offending total.
        total: u128,
    },
    /// The spend would consume time-locked founder tranches.
    FounderTrancheLocked {
        /// Value still locked at the current height.
        locked: Atoms,
        /// `U` after the operation.
        circulating_after: Atoms,
    },
    /// `η` must satisfy `0 < u < v` and `gcd(u, v) = 1`.
    InvalidEmissionRate {
        /// `u`.
        numerator: u64,
        /// `v`.
        denominator: u64,
    },
    /// A deployment parameter is unusable.
    InvalidParams(&'static str),
    /// Heights only move forward.
    HeightRegression {
        /// Current height.
        current: u64,
        /// Requested height.
        requested: u64,
    },
    /// No escrow exists for the job.
    UnknownJob(JobId),
    /// The job already has an escrow.
    JobAlreadyFunded(JobId),
    /// The escrow cannot cover the requested movement.
    InsufficientEscrow {
        /// The job.
        job: JobId,
        /// Escrow balance.
        available: Atoms,
        /// Requested amount.
        required: Atoms,
    },
    /// Tariffs are positive integers.
    ZeroAmount,
    /// Rewards for this epoch were already booked.
    EpochAlreadyBooked(u64),
    /// A sub-ledger disagrees with its category balance.
    Inconsistent(&'static str),
}

impl fmt::Display for LedgerError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            LedgerError::NonZeroSumDelta(s) => write!(f, "delta sums to {s}, not zero"),
            LedgerError::NegativeBalance(c) => write!(f, "{} would become negative", c.symbol()),
            LedgerError::BalanceOverflow(c) => write!(f, "{} would overflow", c.symbol()),
            LedgerError::SupplyViolation { total } => write!(f, "categories sum to {total}, not M"),
            LedgerError::FounderTrancheLocked {
                locked,
                circulating_after,
            } => write!(
                f,
                "U would drop to {} atoms below {} locked founder atoms",
                circulating_after.get(),
                locked.get()
            ),
            LedgerError::InvalidEmissionRate {
                numerator,
                denominator,
            } => {
                write!(f, "invalid emission rate {numerator}/{denominator}")
            }
            LedgerError::InvalidParams(why) => write!(f, "invalid parameters: {why}"),
            LedgerError::HeightRegression { current, requested } => {
                write!(f, "height {requested} is below current height {current}")
            }
            LedgerError::UnknownJob(j) => write!(f, "no escrow for job {}", j.0),
            LedgerError::JobAlreadyFunded(j) => write!(f, "job {} is already funded", j.0),
            LedgerError::InsufficientEscrow {
                job,
                available,
                required,
            } => write!(
                f,
                "job {} escrow {} atoms < required {} atoms",
                job.0,
                available.get(),
                required.get()
            ),
            LedgerError::ZeroAmount => write!(f, "amounts must be positive"),
            LedgerError::EpochAlreadyBooked(e) => write!(f, "epoch {e} already booked"),
            LedgerError::Inconsistent(what) => write!(f, "inconsistent state: {what}"),
        }
    }
}

impl std::error::Error for LedgerError {}

/// The eight-category ledger state with its sub-ledgers.
#[derive(Clone, Debug)]
pub struct Ledger {
    params: LedgerParams,
    height: u64,
    balances: [Atoms; StateCategory::COUNT],
    founder: FounderVesting,
    escrows: BTreeMap<JobId, Atoms>,
    locked: Vec<LockedFee>,
    lots: Vec<FeeLot>,
    segregated_dust: Atoms,
    next_lot: u64,
    last_booked_epoch: Option<u64>,
}

impl Ledger {
    /// Genesis: `U = F` as 1,460 founder tranches, `R_a = M - F` as the Genesis
    /// emission lot, and zero in the other six categories.
    pub fn genesis(params: LedgerParams) -> Result<Self, LedgerError> {
        if params.blocks_per_epoch == 0 {
            return Err(LedgerError::InvalidParams(
                "blocks_per_epoch must be positive",
            ));
        }
        let mut balances = [Atoms::ZERO; StateCategory::COUNT];
        balances[StateCategory::Circulating.index()] = FOUNDER_ALLOCATION;
        balances[StateCategory::EmissionReserve.index()] = GENESIS_EMISSION_RESERVE;
        let ledger = Ledger {
            params,
            height: params.genesis_height,
            balances,
            founder: FounderVesting::new(params.genesis_height, FOUNDER_ALLOCATION),
            escrows: BTreeMap::new(),
            locked: Vec::new(),
            lots: vec![FeeLot::genesis(GENESIS_EMISSION_RESERVE)],
            segregated_dust: Atoms::ZERO,
            next_lot: 1,
            last_booked_epoch: None,
        };
        ledger.check_invariants()?;
        Ok(ledger)
    }

    /// Deployment parameters.
    pub fn params(&self) -> &LedgerParams {
        &self.params
    }
    /// Current height.
    pub fn height(&self) -> u64 {
        self.height
    }
    /// Epoch containing the current height.
    pub fn epoch(&self) -> u64 {
        self.epoch_of(self.height)
    }
    /// Epoch `e(h)` containing `height`.
    pub fn epoch_of(&self, height: u64) -> u64 {
        height.saturating_sub(self.params.genesis_height) / self.params.blocks_per_epoch
    }
    /// Balance of one category.
    pub fn balance(&self, category: StateCategory) -> Atoms {
        self.balances[category.index()]
    }
    /// Sum over all eight categories.
    pub fn total(&self) -> u128 {
        self.balances.iter().map(|b| u128::from(b.get())).sum()
    }
    /// The FounderAllocation schedule.
    pub fn founder_vesting(&self) -> &FounderVesting {
        &self.founder
    }
    /// Lots in `R_a`: the Genesis emission lot followed by MatureFeeLots.
    pub fn fee_lots(&self) -> &[FeeLot] {
        &self.lots
    }
    /// Fees still locked in `R_l`.
    pub fn locked_fees(&self) -> &[LockedFee] {
        &self.locked
    }
    /// Rounding dust permanently segregated in `R_a`.
    pub fn segregated_dust(&self) -> Atoms {
        self.segregated_dust
    }
    /// Escrow balance of a job.
    pub fn escrow(&self, job: JobId) -> Atoms {
        self.escrows.get(&job).copied().unwrap_or(Atoms::ZERO)
    }

    /// Advances the height. This changes lock eligibility only and produces no
    /// balance delta.
    pub fn advance_to(&mut self, height: u64) -> Result<(), LedgerError> {
        if height < self.height {
            return Err(LedgerError::HeightRegression {
                current: self.height,
                requested: height,
            });
        }
        self.height = height;
        Ok(())
    }

    /// `ESCROW_FUND_AND_QUEUE`: `ΔU = -(D + f_queue)`, `ΔE_esc = D`, `ΔR_l = f_queue`.
    /// The queue fee is an irrevocable fee transfer and becomes its own lot.
    pub fn escrow_fund_and_queue(
        &mut self,
        job: JobId,
        deposit: Atoms,
        queue_fee: Atoms,
    ) -> Result<BalanceDelta, LedgerError> {
        if deposit == Atoms::ZERO || queue_fee == Atoms::ZERO {
            return Err(LedgerError::ZeroAmount);
        }
        if self.escrows.contains_key(&job) {
            return Err(LedgerError::JobAlreadyFunded(job));
        }
        let total = i128::from(deposit.get()) + i128::from(queue_fee.get());
        let delta = BalanceDelta::ZERO
            .with(StateCategory::Circulating, -total)
            .with(StateCategory::Escrow, i128::from(deposit.get()))
            .with(StateCategory::LockedBaseFees, i128::from(queue_fee.get()));
        let staged = self.stage(&delta)?;
        self.balances = staged;
        self.escrows.insert(job, deposit);
        self.lock_fee(queue_fee);
        Ok(delta)
    }

    /// `JOB_ADMIT_AND_LEASE`: `ΔE_esc = -F_base`, `ΔR_l = F_base`.
    ///
    /// The Base Fee moves irretrievably into `R_l[e(h_admit) + L]`; no operation
    /// ever moves value from `R_l` back to an escrow or a client (eq. 7.21).
    pub fn job_admit_and_lease(
        &mut self,
        job: JobId,
        base_fee: Atoms,
    ) -> Result<BalanceDelta, LedgerError> {
        if base_fee == Atoms::ZERO {
            return Err(LedgerError::ZeroAmount);
        }
        self.require_escrow(job, base_fee)?;
        let delta = BalanceDelta::transfer(
            StateCategory::Escrow,
            StateCategory::LockedBaseFees,
            base_fee,
        );
        let staged = self.stage(&delta)?;
        self.balances = staged;
        self.debit_escrow(job, base_fee);
        self.lock_fee(base_fee);
        Ok(delta)
    }

    /// `RESULT_SETTLE`: `ΔE_esc = -(v_exec + v_DA + v_ref)`, `ΔU = v_exec + v_DA + v_ref`.
    pub fn result_settle(
        &mut self,
        job: JobId,
        v_exec: Atoms,
        v_da: Atoms,
        v_ref: Atoms,
    ) -> Result<BalanceDelta, LedgerError> {
        let total = [v_exec, v_da, v_ref]
            .into_iter()
            .try_fold(Atoms::ZERO, Atoms::checked_add)
            .ok_or(LedgerError::BalanceOverflow(StateCategory::Escrow))?;
        self.require_escrow(job, total)?;
        let delta =
            BalanceDelta::transfer(StateCategory::Escrow, StateCategory::Circulating, total);
        let staged = self.stage(&delta)?;
        self.balances = staged;
        self.debit_escrow(job, total);
        Ok(delta)
    }

    /// `RESERVE_MATURE`: every locked fee whose maturity epoch has arrived moves
    /// `R_l -> R_a` as a new, independent MatureFeeLot (`ΔR_l = -v_mat`, `ΔR_a = v_mat`).
    pub fn reserve_mature(&mut self) -> Result<(BalanceDelta, Vec<LotId>), LedgerError> {
        let epoch = self.epoch();
        let (ready, waiting): (Vec<LockedFee>, Vec<LockedFee>) =
            self.locked.iter().partition(|f| f.maturity_epoch <= epoch);
        let v_mat = Atoms(ready.iter().map(|f| f.amount.get()).sum());
        let delta = BalanceDelta::transfer(
            StateCategory::LockedBaseFees,
            StateCategory::EmissionReserve,
            v_mat,
        );
        let staged = self.stage(&delta)?;
        self.balances = staged;
        self.locked = waiting;
        let ids = ready.iter().map(|f| f.lot).collect();
        self.lots.extend(
            ready
                .iter()
                .map(|f| FeeLot::matured(f.lot, f.amount, f.paid_epoch, f.maturity_epoch)),
        );
        Ok((delta, ids))
    }

    /// Epoch budget: every eligible lot settles against the mature share
    /// weights; booked payouts move `R_a -> P_pending` (`ΔR_a = -P_e`,
    /// `ΔP_pending = P_e`) and the dust `r_e` stays segregated in `R_a`.
    /// Each epoch is booked at most once.
    pub fn book_epoch(
        &mut self,
        weights: &[ShareWeight],
    ) -> Result<(BalanceDelta, EpochPayout), LedgerError> {
        let epoch = self.epoch();
        if self.last_booked_epoch.is_some_and(|last| last >= epoch) {
            return Err(LedgerError::EpochAlreadyBooked(epoch));
        }
        let rate = self.params.emission_rate;
        let mut lots = self.lots.clone();
        let mut per_share = vec![Atoms::ZERO; weights.len()];
        let mut details = Vec::new();
        for lot in lots
            .iter_mut()
            .filter(|l| l.maturity_epoch <= epoch && l.principal > Atoms::ZERO)
        {
            let payout = lot.settle_epoch(&rate, weights);
            for (acc, r) in per_share.iter_mut().zip(&payout.rewards) {
                *acc = Atoms(acc.get() + r.get());
            }
            details.push(payout);
        }
        let sum = |f: fn(&LotEpochPayout) -> Atoms| Atoms(details.iter().map(|d| f(d).get()).sum());
        let payout = EpochPayout {
            epoch,
            budget: sum(|d| d.budget),
            booked: sum(|d| d.booked),
            dust: sum(|d| d.dust),
            per_share,
            lots: details,
        };
        let delta = BalanceDelta::transfer(
            StateCategory::EmissionReserve,
            StateCategory::PendingRewards,
            payout.booked,
        );
        let staged = self.stage(&delta)?;
        self.balances = staged;
        self.lots = lots;
        self.segregated_dust = Atoms(self.segregated_dust.get() + payout.dust.get());
        self.last_booked_epoch = Some(epoch);
        Ok((delta, payout))
    }

    /// `FW_REWARD`: `ΔP_pending = -v_pay`, `ΔU = v_pay`.
    pub fn fw_reward(&mut self, amount: Atoms) -> Result<BalanceDelta, LedgerError> {
        let delta = BalanceDelta::transfer(
            StateCategory::PendingRewards,
            StateCategory::Circulating,
            amount,
        );
        self.balances = self.stage(&delta)?;
        Ok(delta)
    }

    /// Verifies the supply invariant and that every sub-ledger matches its category.
    pub fn check_invariants(&self) -> Result<(), LedgerError> {
        if self.total() != u128::from(TOTAL_SUPPLY.get()) {
            return Err(LedgerError::SupplyViolation {
                total: self.total(),
            });
        }
        let reserve =
            self.segregated_dust.get() + self.lots.iter().map(|l| l.principal.get()).sum::<u64>();
        if reserve != self.balance(StateCategory::EmissionReserve).get() {
            return Err(LedgerError::Inconsistent(
                "R_a != dust + sum of lot principals",
            ));
        }
        let locked: u64 = self.locked.iter().map(|f| f.amount.get()).sum();
        if locked != self.balance(StateCategory::LockedBaseFees).get() {
            return Err(LedgerError::Inconsistent("R_l != sum of locked fees"));
        }
        let escrowed: u64 = self.escrows.values().map(|a| a.get()).sum();
        if escrowed != self.balance(StateCategory::Escrow).get() {
            return Err(LedgerError::Inconsistent("E_esc != sum of job escrows"));
        }
        if self.lots.iter().any(|l| {
            l.principal.get() + l.booked_total.get() + l.dust_total.get() != l.source_fee.get()
        }) {
            return Err(LedgerError::Inconsistent(
                "lot principal + booked + dust != source fee",
            ));
        }
        if self.balance(StateCategory::Circulating) < self.founder.locked_at(self.height) {
            return Err(LedgerError::Inconsistent("U below locked founder tranches"));
        }
        Ok(())
    }

    /// Validates a delta against the current state without mutating it:
    /// zero sum (eq. 7.13), non-negativity, supply `M` (eq. 7.19) and the
    /// founder time locks. Callers commit the returned balances together with
    /// their sub-ledger updates, which makes every operation all-or-nothing.
    fn stage(&self, delta: &BalanceDelta) -> Result<[Atoms; StateCategory::COUNT], LedgerError> {
        if delta.sum() != 0 {
            return Err(LedgerError::NonZeroSumDelta(delta.sum()));
        }
        let mut next = self.balances;
        for category in StateCategory::ALL {
            let value = i128::from(next[category.index()].get()) + delta.get(category);
            if value < 0 {
                return Err(LedgerError::NegativeBalance(category));
            }
            next[category.index()] =
                Atoms(u64::try_from(value).map_err(|_| LedgerError::BalanceOverflow(category))?);
        }
        let total: u128 = next.iter().map(|b| u128::from(b.get())).sum();
        if total != u128::from(TOTAL_SUPPLY.get()) {
            return Err(LedgerError::SupplyViolation { total });
        }
        let locked = self.founder.locked_at(self.height);
        let circulating_after = next[StateCategory::Circulating.index()];
        if circulating_after < locked {
            return Err(LedgerError::FounderTrancheLocked {
                locked,
                circulating_after,
            });
        }
        Ok(next)
    }

    fn require_escrow(&self, job: JobId, required: Atoms) -> Result<(), LedgerError> {
        let available = *self.escrows.get(&job).ok_or(LedgerError::UnknownJob(job))?;
        if available < required {
            return Err(LedgerError::InsufficientEscrow {
                job,
                available,
                required,
            });
        }
        Ok(())
    }

    fn debit_escrow(&mut self, job: JobId, amount: Atoms) {
        if let Some(balance) = self.escrows.get_mut(&job) {
            *balance = Atoms(balance.get() - amount.get());
            if *balance == Atoms::ZERO {
                self.escrows.remove(&job);
            }
        }
    }

    fn lock_fee(&mut self, amount: Atoms) {
        let paid_epoch = self.epoch();
        self.locked.push(LockedFee {
            lot: LotId(self.next_lot),
            amount,
            paid_epoch,
            maturity_epoch: paid_epoch + self.params.maturation_delay + 1,
        });
        self.next_lot += 1;
    }
}

fn gcd(mut a: u64, mut b: u64) -> u64 {
    while b != 0 {
        (a, b) = (b, a % b);
    }
    a
}

    fn zero_sum_delta_rule() {
        let mut ledger = Ledger::genesis(LedgerParams::poc()).unwrap();
        ledger.advance_to(VESTING_BLOCKS).unwrap(); // everything vested
        let before = ledger.balances;

        // Minting one atom is rejected and changes nothing.
        let mint = BalanceDelta::ZERO.with(StateCategory::Circulating, 1);
        assert_eq!(ledger.stage(&mint), Err(LedgerError::NonZeroSumDelta(1)));
        // A zero-sum delta that overdraws a category is rejected too.
        let overdraw =
            BalanceDelta::transfer(StateCategory::Bonds, StateCategory::Circulating, Atoms(1));
        assert_eq!(
            ledger.stage(&overdraw),
            Err(LedgerError::NegativeBalance(StateCategory::Bonds))
        );
        assert_eq!(ledger.balances, before);

        // Shield, bond and bridge value: every category moves, the sum does not.
        let spread = BalanceDelta::ZERO
            .with(StateCategory::Circulating, -3_000)
            .with(StateCategory::ShieldedPool, 1_000)
            .with(StateCategory::Bonds, 1_500)
            .with(StateCategory::L2Vaults, 500);
        assert_eq!(spread.sum(), 0);

    #[test]
    fn fee_lot_pays_by_integer_division_and_segregates_dust() {
        let rate = EmissionRate::new(1, 1_000).unwrap();
        let mut lot = FeeLot::matured(LotId(9), Atoms(1_000_000_000), 0, 3);
        let first = lot.settle_epoch(&rate, &[1, 1, 1]);
        assert_eq!(first.budget, Atoms(1_000_000));
        assert_eq!(first.rewards, vec![Atoms(333_333); 3]);
        assert_eq!((first.booked, first.dust), (Atoms(999_999), Atoms(1)));
        assert_eq!(lot.principal(), Atoms(999_000_000));

        // Dust never returns to principal: the next budget is ceil(η * 999,000,000).
        let second = lot.settle_epoch(&rate, &[2, 5]);
        assert_eq!(second.budget, Atoms(999_000));

        assert_eq!(rate.budget(Atoms(1_001)), Atoms(2));
        assert_eq!(rate.budget(Atoms(1)), Atoms(1));
        assert_eq!(rate.budget(Atoms::ZERO), Atoms::ZERO);

//! Plauthium proof of concept: one mock block transition across all five modules.
//!
//! a) Genesis ledger: Emission Reserve and Founder Allocation.
//! b) Fresh Work: expand a seed into two 512 x 512 INT8 tiles, compute the
//!    exact INT32 product and certify it with six Freivalds rounds.
//! c) Scheduler: sequential D'Hondt lease allocation over discrete ShareIDs.
//! d) Ledger: lock a Base Fee, mature it into an independent fee lot and pay
//!    Fresh Work rewards from it with segregated rounding dust.
//! e) ASCII summary.
//!
//! Hashes, XOFs and signatures are placeholders (see the `primitives` crate).

use std::error::Error;

use ledger::{
    BLOCKS_PER_DAY, FOUNDER_ALLOCATION, JobId, Ledger, LedgerParams, StateCategory,
    VESTING_TRANCHES,
};
use primitives::{Atoms, DomainSeparatedHash, Mersenne61, TOTAL_SUPPLY};
use scheduler::{Allocation, OwnerId, ShareID, dhondt_allocation};
use scivm::{CONSENSUS_WEIGHT, OzakiIIProfile, RoundingMode};
use work::{
    ACCUMULATION_BOUND, CLASSICAL_MACS, DEFAULT_FREIVALDS_ROUNDS, FREIVALDS_FIELD_MULS,
    FreshWorkContext, FreshWorkIds, TILE_DIM, verify_freivalds_rounds,
};

/// Height of the block this PoC seals.
const BLOCK_HEIGHT: u64 = 1;
/// Execution slots (ready tiles) offered to the scheduler in this batch.
const LEASE_SLOTS: usize = 8;

type PocResult<T> = Result<T, Box<dyn Error>>;

fn main() -> PocResult<()> {
    println!("Plauthium PoC - mock block transition");
    println!("(hash/XOF/signatures are non-cryptographic placeholders)\n");

    // ---------------------------------------------------------------- (a)
    section("a", "Genesis ledger");
    let mut ledger = Ledger::genesis(LedgerParams::poc())?;
    print_categories(&ledger);
    let vesting = *ledger.founder_vesting();
    let first = vesting.tranche(1).ok_or("missing tranche 1")?;
    let last = vesting
        .tranche(VESTING_TRANCHES)
        .ok_or("missing last tranche")?;
    println!(
        "  FounderAllocation: {} tranches inside U; tranche 1 = {} atoms @ h={}, tranche {} @ h={}",
        VESTING_TRANCHES,
        group(first.value.get()),
        first.unlock_height,
        last.day,
        group(last.unlock_height),
    );
    println!(
        "  EmissionReserve:   Genesis lot c_G with A = {} atoms\n",
        group(ledger.fee_lots()[0].principal().get())
    );

    // ---------------------------------------------------------------- (b)
    section(
        "b",
        "Fresh Work: seed -> 512x512 INT8 tiles -> INT32 product",
    );
    let context = mock_context(BLOCK_HEIGHT, 0);
    let instance = context.instance();
    println!(
        "  seed = H512(FW/INPUT/v2, x)       {}..",
        instance.seed.short_hex(16)
    );
    println!(
        "  X, Y = Map(XOF(FW/A|FW/B, seed))  {TILE_DIM}x{TILE_DIM}, entries in [-127, 127], max|X| = {}",
        instance.x.max_abs()
    );
    let product = instance.x.product(&instance.y);
    println!(
        "  Z = XY ({} MACs)         max|Z| = {} <= {}",
        group(CLASSICAL_MACS),
        group(product.max_abs() as u64),
        group(ACCUMULATION_BOUND as u64)
    );
    let accepted =
        verify_freivalds_rounds(&instance.x, &instance.y, &product, DEFAULT_FREIVALDS_ROUNDS);
    let mut forged = product.clone();
    let entry = forged.get(17, 42);
    forged.set(17, 42, if entry > 0 { entry - 1 } else { entry + 1 })?;
    let forged_rejected =
        !verify_freivalds_rounds(&instance.x, &instance.y, &forged, DEFAULT_FREIVALDS_ROUNDS);
    ensure(accepted, "Freivalds rejected the honest product")?;
    ensure(forged_rejected, "Freivalds accepted a forged product")?;
    println!(
        "  Freivalds r_F = {DEFAULT_FREIVALDS_ROUNDS} over F_p (p = {}): honest ACCEPT, forged REJECT",
        Mersenne61::P
    );
    let ids = FreshWorkIds::derive(&context, &product);
    println!("  outputRoot {}..", ids.output_root.short_hex(16));
    println!("  WorkID     {}..  (= ShareID)", ids.work_id.short_hex(16));
    println!("  BlockID    {}..", ids.block_id.short_hex(16));
    println!(
        "  theta_S    {}..  theta_B {}..\n",
        ids.share_ticket.short_hex(8),
        ids.seal_ticket.short_hex(8)
    );

    // ---------------------------------------------------------------- (c)
    section("c", "Scheduler: sequential D'Hondt over discrete ShareIDs");
    let shares = mock_cohort(ids.work_id);
    let allocation = dhondt_allocation(&shares, LEASE_SLOTS)?;
    ensure(
        allocation.assigned() == LEASE_SLOTS,
        "D'Hondt left slots unassigned",
    )?;
    print_allocation(&shares, &allocation);

    // ---------------------------------------------------------------- (d)
    section("d", "Ledger: Base Fee lock -> maturation -> fee-lot payout");
    let weights: Vec<u64> = shares.iter().map(|s| s.weight).collect();

    // Epoch 0 rewards from the Genesis lot give miners spendable outputs in U.
    let (_, genesis_epoch) = ledger.book_epoch(&weights)?;
    ledger.fw_reward(genesis_epoch.booked)?;
    step(
        &ledger,
        "FW_REWARD (epoch 0, Genesis lot)",
        format!(
            "{} paid, {} atoms dust",
            genesis_epoch.booked,
            genesis_epoch.dust.get()
        ),
    )?;

    // The commercial job runs on the SciVM under a bound Ozaki-II profile.
    let profile = OzakiIIProfile {
        version: 1,
        moduli: vec![255, 254, 253, 251, 247, 241, 239, 233],
        inner_dim: TILE_DIM as u64,
        bound_x: 1 << 23,
        bound_y: 1 << 23,
        rounding: RoundingMode::NearestTiesToEven,
    };
    profile.validate()?;
    let (xs, ys) = mock_scientific_operands();
    let dot = profile.exact_dot(&xs, &ys)?;
    let direct: i128 = xs
        .iter()
        .zip(&ys)
        .map(|(&a, &b)| a as i128 * b as i128)
        .sum();
    ensure(dot == direct, "Ozaki-II reconstruction mismatch")?;
    println!(
        "  SciVM Ozaki-II: {} moduli, Q ~ 2^{:.1} > 2*B_rec; exact dot = {} (consensus weight {CONSENSUS_WEIGHT})",
        profile.moduli.len(),
        (profile.reconstruction_modulus()? as f64).log2(),
        dot
    );

    // Worked budget of eq. (7.17): D = F_base + v_exec + v_DA + v_ref = 100 PLAU.
    let job = JobId(1);
    let (deposit, queue_fee, base_fee) = (plau(100)?, plau(1)?, plau(10)?);
    ledger.escrow_fund_and_queue(job, deposit, queue_fee)?;
    step(
        &ledger,
        "ESCROW_FUND_AND_QUEUE",
        format!("D = {deposit}, f_queue = {queue_fee}"),
    )?;
    ledger.job_admit_and_lease(job, base_fee)?;
    step(
        &ledger,
        "JOB_ADMIT_AND_LEASE",
        format!("Base Fee {base_fee} -> R_l (irretrievable)"),
    )?;
    ledger.result_settle(job, plau(80)?, plau(5)?, plau(5)?)?;
    step(
        &ledger,
        "RESULT_SETTLE",
        "v_exec 80 + v_DA 5 + v_ref 5 PLAU -> U".to_string(),
    )?;

    let maturity_epoch = ledger
        .locked_fees()
        .iter()
        .map(|f| f.maturity_epoch)
        .max()
        .ok_or("no locked fees")?;
    let maturity_height =
        ledger.params().genesis_height + maturity_epoch * ledger.params().blocks_per_epoch;
    ledger.advance_to(maturity_height)?;
    let (_, matured) = ledger.reserve_mature()?;
    step(
        &ledger,
        "RESERVE_MATURE",
        format!(
            "h={maturity_height} (epoch {maturity_epoch}): {} lots R_l -> R_a",
            matured.len()
        ),
    )?;

    let (_, epoch) = ledger.book_epoch(&weights)?;
    ledger.fw_reward(epoch.booked)?;
    step(
        &ledger,
        "FW_REWARD (epoch with fee lots)",
        format!("{} paid from {} lots", epoch.booked, epoch.lots.len()),
    )?;
    for lot in epoch.lots.iter().filter(|l| matured.contains(&l.lot)) {
        let source = ledger
            .fee_lots()
            .iter()
            .find(|l| l.id() == lot.lot)
            .map_or(Atoms::ZERO, |l| l.source_fee());
        println!(
            "    fee lot #{} (F_c = {source}): budget {} atoms = {} booked + {} dust (segregated)",
            lot.lot.0,
            group(lot.budget.get()),
            group(lot.booked.get()),
            lot.dust.get()
        );
    }
    ledger.check_invariants()?;
    println!();
    print_categories(&ledger);

    // ---------------------------------------------------------------- (e)
    print_summary(&ledger, &ids, &allocation, matured.len(), epoch.dust)?;
    Ok(())
}

fn tag(name: &str) -> DomainSeparatedHash {
    DomainSeparatedHash::new("POC/MOCK/v2", &[name.as_bytes()])
}

fn mock_context(height: u64, attempt_nonce: u64) -> FreshWorkContext {
    FreshWorkContext {
        chain_id: 0x504c_4155, // "PLAU"
        genesis_id: tag("genesis"),
        protocol_version: 2,
        crypto_suite_id: 1,
        program_id: tag("fresh-work-program"),
        air_id: tag("int8-tile-air"),
        generator_version: 2,
        parent_canonical_id: tag("block-0"),
        stable_anchor_id: tag("anchor-0"),
        height,
        miner_policy_digest: tag("miner-policy"),
        payout_policy_digest: tag("payout-policy"),
        attempt_nonce,
        body_core_root: tag("body-core-1"),
        parent_derived_difficulty_context: tag("difficulty-0"),
        time_bucket: height * ledger::BLOCK_INTERVAL_SECONDS,
    }
}

/// The fresh share plus five mature shares from other miners. Owner 2 holds
/// two ShareIDs: its slots are simply the sum over them.
fn mock_cohort(fresh: DomainSeparatedHash) -> Vec<ShareID> {
    let mut shares = vec![ShareID {
        id: fresh,
        weight: 1_000_003,
        owner: OwnerId(1),
    }];
    for (i, (weight, owner)) in [
        (2_400_017u64, 2u64),
        (700_001, 2),
        (1_600_009, 3),
        (300_007, 4),
        (999_983, 5),
    ]
    .into_iter()
    .enumerate()
    {
        shares.push(ShareID {
            id: tag(&format!("mature-share-{i}")),
            weight,
            owner: OwnerId(owner),
        });
    }
    shares
}

/// Deterministic scaled integer operands with |x|, |y| < 2^23.
fn mock_scientific_operands() -> (Vec<f64>, Vec<f64>) {
    let operands = |mul: u64| -> Vec<f64> {
        (0..TILE_DIM as u64)
            .map(|i| ((i.wrapping_mul(mul) >> 7) % (1 << 24)) as f64 - (1 << 23) as f64)
            .collect()
    };
    (operands(2_654_435_761), operands(40_503))
}

fn plau(n: u64) -> PocResult<Atoms> {
    Atoms::from_plau(n).ok_or_else(|| "PLAU amount overflows".into())
}

fn ensure(condition: bool, message: &str) -> PocResult<()> {
    if condition {
        Ok(())
    } else {
        Err(message.into())
    }
}

fn section(letter: &str, title: &str) {
    println!("({letter}) {title}");
}

/// Reports one operation and re-checks the supply after it.
fn step(ledger: &Ledger, op: &str, detail: String) -> PocResult<()> {
    ensure(
        ledger.total() == u128::from(TOTAL_SUPPLY.get()),
        "supply changed",
    )?;
    println!("  {op:<32} {detail}");
    Ok(())
}

fn group(n: u64) -> String {
    let digits = n.to_string();
    let mut out = String::with_capacity(digits.len() + digits.len() / 3);
    for (i, c) in digits.chars().enumerate() {
        if i > 0 && (digits.len() - i).is_multiple_of(3) {
            out.push(',');
        }
        out.push(c);
    }
    out
}

fn print_categories(ledger: &Ledger) {
    let rule = "  +-----------------+-----------+---------------------------+";
    println!("{rule}");
    println!(
        "  | {:<15} | {:<9} | {:>25} |",
        "Category", "Symbol", "Balance (atoms)"
    );
    println!("{rule}");
    for category in StateCategory::ALL {
        println!(
            "  | {:<15} | {:<9} | {:>25} |",
            format!("{category:?}"),
            category.symbol(),
            group(ledger.balance(category).get())
        );
    }
    println!("{rule}");
    let total = ledger.total() as u64;
    let verdict = if u128::from(total) == u128::from(TOTAL_SUPPLY.get()) {
        "== M"
    } else {
        "!= M"
    };
    println!(
        "  | {:<15} | {:<9} | {:>25} |",
        "SUM",
        verdict,
        group(total)
    );
    println!("{rule}\n");
}

fn print_allocation(shares: &[ShareID], allocation: &Allocation) {
    let rule = "  +----+--------------------+-------+-----------+-------+";
    println!("{rule}");
    println!("  | #  | ShareID            | Owner | Weight    | Slots |");
    println!("{rule}");
    for (i, (share, slots)) in shares.iter().zip(&allocation.per_share).enumerate() {
        println!(
            "  | {i:<2} | {}.. | {:<5} | {:>9} | {slots:>5} |",
            share.id.short_hex(8),
            share.owner.0,
            group(share.weight)
        );
    }
    println!("{rule}");
    let order: Vec<String> = allocation.slots.iter().map(|s| s.to_string()).collect();
    println!("  lease order (slot -> share #): {}\n", order.join(" "));
}

fn print_summary(
    ledger: &Ledger,
    ids: &FreshWorkIds,
    allocation: &Allocation,
    matured_lots: usize,
    epoch_dust: Atoms,
) -> PocResult<()> {
    const WIDTH: usize = 72;
    let border = format!("+{}+", "=".repeat(WIDTH));
    let rule = format!("+{}+", "-".repeat(WIDTH));
    let line = |key: &str, value: String| println!("| {key:<20}: {value:<w$}|", w = WIDTH - 23);
    let distinct = allocation.per_share.iter().filter(|&&a| a > 0).count();
    let vested = ledger
        .founder_vesting()
        .completed_intervals(ledger.height());

    println!("{border}");
    println!("|{:^WIDTH$}|", "PLAUTHIUM BLOCK EXECUTED SUCCESSFULLY");
    println!("{border}");
    line(
        "Block height",
        format!("{BLOCK_HEIGHT} (state advanced to h={})", ledger.height()),
    );
    line("BlockID", format!("{}..", ids.block_id.short_hex(16)));
    line(
        "WorkID = ShareID",
        format!("{}..", ids.work_id.short_hex(16)),
    );
    line(
        "Work unit",
        format!(
            "{TILE_DIM}x{TILE_DIM} INT8 -> INT32, |Z| <= {}",
            group(ACCUMULATION_BOUND as u64)
        ),
    );
    line(
        "Freivalds",
        format!(
            "ACCEPT, r_F = {DEFAULT_FREIVALDS_ROUNDS}, {} F_p mults (9/256)",
            group(FREIVALDS_FIELD_MULS)
        ),
    );
    line(
        "D'Hondt leases",
        format!(
            "{}/{LEASE_SLOTS} slots -> {distinct} discrete ShareIDs",
            allocation.assigned()
        ),
    );
    line(
        "Fee lots matured",
        format!("{matured_lots} (queue fee + Base Fee), R_l -> R_a"),
    );
    line(
        "Dust this epoch",
        format!(
            "{} atoms (total segregated {})",
            epoch_dust.get(),
            ledger.segregated_dust().get()
        ),
    );
    line(
        "Founder vesting",
        format!(
            "{vested}/{VESTING_TRANCHES} tranches unlocked ({} days x {BLOCKS_PER_DAY} blocks)",
            vested
        ),
    );
    line(
        "Founder locked",
        format!(
            "{} of {} atoms",
            group(ledger.founder_vesting().locked_at(ledger.height()).get()),
            group(FOUNDER_ALLOCATION.get())
        ),
    );
    println!("{rule}");
    for category in StateCategory::ALL {
        line(
            category.symbol(),
            format!(
                "{:<16}{:>25}",
                format!("{category:?}"),
                group(ledger.balance(category).get())
            ),
        );
    }
    println!("{rule}");
    let total = ledger.total();
    ensure(
        total == u128::from(TOTAL_SUPPLY.get()),
        "supply invariant violated",
    )?;
    line(
        "SUM (== M)",
        format!("{:<16}{:>25}  OK", "", group(total as u64)),
    );
    println!("{border}");
    Ok(())
}
